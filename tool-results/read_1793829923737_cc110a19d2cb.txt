   346→/* ===== Panel A — Sector Performance (Real Data) ===== */
   347→function SectorPerformancePanel({
   348→  sectorAggs,
   349→  onOpenCompany,
   350→  isPremium,
   351→  ar,
   352→  lang,
   353→  onViewChange,
   354→}: {
   355→  sectorAggs: SectorAgg[]
   356→  onOpenCompany: (id: string) => void
   357→  isPremium: boolean
   358→  ar: boolean
   359→  lang: "en" | "ar"
   360→  onViewChange?: (v: ViewKey) => void
   361→}) {
   362→  const router = useRouter()
   363→
   364→  const rowsWithData = sectorAggs
   365→    .filter((s) => s.dayChangePct !== null)
   366→    .sort((a, b) => (b.dayChangePct ?? -Infinity) - (a.dayChangePct ?? -Infinity))
   367→
   368→  return (
   369→    <div className="lg:col-span-4">
   370→      <Card className="border-border/60 p-0">
   371→        <CardHeader className="pb-3 px-5">
   372→          <div className="flex items-center justify-between gap-2">
   373→            <CardTitle className="text-sm font-semibold flex items-center gap-2">
   374→              <Gauge className="h-3.5 w-3.5 text-primary" />
   375→              {ar ? "أداء القطاعات (يوم)" : "Sector Performance (1D)"}
   376→            </CardTitle>
   377→            <Button variant="ghost" size="sm" className="h-6 px-2 text-[10px]" onClick={() => router.push("/sectors")}>
   378→              {ar ? "عرض الكل" : "Show all"}
   379→            </Button>
   380→          </div>
   381→        </CardHeader>
   382→        <CardContent className="p-0">
   383→          <div className="space-y-1.5 p-4">
   384→            {rowsWithData.length > 0 ? (
   385→              rowsWithData.slice(0, 5).map((row, i) => {
   386→                const change = row.dayChangePct
   387→                const positive = change !== null && change >= 0
   388→                const widthPct = change !== null ? Math.min(Math.abs(change) * 10, 100) : 0
   389→                return (
   390→                  <div
   391→                    key={row.sector}
   392→                    onClick={() => router.push("/sectors/" + encodeURIComponent(row.sector))}
   393→                    className="group flex items-center justify-between gap-2 rounded-xl px-3 py-2.5 text-start transition-all duration-200 hover:bg-primary/5 hover:border-primary/30 border border-transparent cursor-pointer"
   394→                  >
   395→                    <div className="flex items-center gap-2 min-w-0">
   396→                      <span className="flex h-7 w-7 shrink-0 items-center justify-center rounded-lg bg-primary/10 text-primary text-xs font-bold">
   397→                        {i + 1}
   398→                      </span>
   399→                      <span className="text-sm font-medium truncate">{sectorLabel(row.sector, lang)}</span>
   400→                    </div>
   401→                    <div className="flex-1 h-[8px] mx-3 overflow-hidden rounded-full bg-muted/40">
   402→                      <div
   403→                        className="h-full rounded-full transition-all duration-500"
   404→                        style={{
   405→                          width: widthPct + "%",
   406→                          background: positive
   407→                            ? "linear-gradient(90deg, var(--positive) 0%, #22c55e 100%)"
   408→                            : "linear-gradient(90deg, var(--negative) 0%, #ef4444 100%)",
   409→                        }}
   410→                      />
   411→                    </div>
   412→                    <span className={`text-sm font-bold tabular shrink-0 flex items-center gap-1 ${positive ? "text-positive" : change !== null ? "text-negative" : "text-muted-foreground"}`}>
   413→                      {change !== null ? (
   414→                        <>
   415→                          {positive ? <TrendingUp className="h-3.5 w-3.5" /> : <TrendingDown className="h-3.5 w-3.5" />}
   416→                          <span dir="ltr">{change >= 0 ? "+" : ""}{change.toFixed(1)}%</span>
   417→                        </>
   418→                      ) : "—"}
   419→                    </span>
   420→                  </div>
   421→                )
   422→              })
   423→            ) : (
   424→              <div className="text-center text-xs text-muted-foreground py-4">
   425→                {ar ? "لا توجد أسعار قطاعية متاحة بعد" : "No sector price data available yet"}
   426→              </div>
   427→            )}
   428→          </div>
   429→          <p className="px-4 pb-3 text-[10px] text-muted-foreground">
   430→            {ar ? "تغير السعر المرجّح بالتساوي · متأخر 15 دقيقة" : "Equal-weighted price change · 15-min delayed"}
   431→          </p>
   432→        </CardContent>
   433→      </Card>
   434→    </div>
   435→  )
   436→}

   440→  const advancing = breadth?.advancing ?? 0
   441→  const declining = breadth?.declining ?? 0
   442→  const unchanged = breadth?.unchanged ?? 0
   443→  const tradedValue = breadth?.tradedValue
   444→  const asOf = breadth?.asOf
   445→
   446→  const cells = [
   447→    { label: ar ? "منخفضة" : "Declining", value: String(declining), color: "text-negative", dot: "bg-negative" },
   448→    { label: ar ? "مستقرة" : "Unchanged", value: String(unchanged), color: "text-muted-foreground", dot: "bg-muted-foreground" },
   449→    { label: ar ? "صاعدة" : "Advancing", value: String(advancing), color: "text-positive", dot: "bg-positive" },
   450→    {
   451→      label: ar ? "قيمة التداول" : "Traded Value",
   452→      value: tradedValue !== null && tradedValue !== undefined ? formatEgp(tradedValue) : (ar ? "غير متاح" : "N/A"),
   453→      color: "text-gold",
   454→      dot: "bg-gold/20 text-gold",
   455→    },
   456→  ]
   457→
