    47→function deriveGrossProfit(v: Values): Values {
    48→  if (v["REVENUE"] !== undefined && v["COGS"] !== undefined && v["GROSS_PROFIT"] === undefined) {
    49→    return { ...v, GROSS_PROFIT: v["REVENUE"] - v["COGS"] }
    50→  }
    51→  return v
    52→}
