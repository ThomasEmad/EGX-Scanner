   152→async function fetchText(url) {
   153→  const controller = new AbortController()
   154→  const timer = setTimeout(() => controller.abort(), TIMEOUT_MS)
   155→  try {
   156→    const res = await fetch(url, {
   157→      headers: { "user-agent": USER_AGENT, accept: "text/html,application/xhtml+xml" },
   158→      signal: controller.signal,
   159→    })
   160→    if (res.status === 404) return { status: 404, html: null }
   161→    if (!res.ok) throw new Error(`HTTP ${res.status}`)
   162→    return { status: 200, html: await res.text() }
   163→  } finally {
   164→    clearTimeout(timer)
   165→  }
   166→}

   168→async function fetchWithRetry(url, attempts = 3) {
   169→  for (let i = 1; i <= attempts; i++) {
   170→    try {
   171→      return await fetchText(url)
   172→    } catch (e) {
   173→      if (i === attempts) return { status: 0, error: String(e?.message ?? e) }
   174→      await new Promise((r) => setTimeout(r, 400 * i))
   175→    }
   176→  }
   177→}

   179→async function mapLimit(items, limit, fn) {
   180→  const out = new Array(items.length)
   181→  let next = 0
   182→  let done = 0
   183→  async function worker() {
   184→    while (true) {
   185→      const i = next++
   186→      if (i >= items.length) return
   187→      out[i] = await fn(items[i])
   188→      done++
   189→      if (done % 50 === 0) log(`  progress ${done}/${items.length}`)
   190→      await new Promise((r) => setTimeout(r, STAGGER_MS))
   191→    }
   192→  }
   193→  await Promise.all(Array.from({ length: Math.min(limit, items.length) }, worker))
   194→  return out
   195→}

   202→function extractPivots(html) {
   203→  const text = unescapeNextFlight(html)
   204→  const start = text.indexOf('[{"period":"')
   205→  if (start < 0) return null
   206→  let depth = 0
   207→  let inString = false
   208→  let escaped = false
   209→  let end = -1
   210→  for (let i = start; i < text.length; i++) {
   211→    const ch = text[i]
   212→    if (inString) {
   213→      if (escaped) escaped = false
   214→      else if (ch === "\\") escaped = true
   215→      else if (ch === '"') inString = false
   216→    } else {
   217→      if (ch === '"') inString = true
   218→      else if (ch === "[") depth++
   219→      else if (ch === "]") {
   220→        depth--
   221→        if (depth === 0) { end = i + 1; break }
   222→      }
   223→    }
   224→  }
   225→  if (end < 0) return null
   226→  try {
   227→    const pivots = JSON.parse(text.slice(start, end))
   228→    if (!Array.isArray(pivots)) return null
   229→    return pivots.filter((p) => p && typeof p.period === "string" && Array.isArray(p.rows))
   230→  } catch {
   231→    return null
   232→  }
   233→}

   248→function periodHrefPeriods(html, ticker) {
   249→  const escapedTicker = ticker.replace(/[.*+?^${}()|[\]\\]/g, "\\$")
   250→  const re = new RegExp(`href="/company/${escapedTicker}/(fy|q[1-4])-(\\d{4})"`, "g")
   251→  const out = new Set()
   252→  let m
   253→  while ((m = re.exec(html))) out.add(`${m[1]}-${m[2]}`)
   254→  return [...out]
   255→}

   257→// Statement page: metric blocks in the flight payload.
   258→//   hero card : ["$","div","{id}",{"className":"card ...
   259→//   table row : ["$","div","{id}",{"className":"grid grid-cols... (current, prior, delta)
   260→function parseStatementPage(html, ticker) {
   261→  // each push wraps one JS string literal; match it atomically so an escaped
   262→  // quote or an inner "]" never truncates the chunk
   263→  const chunks = [...html.matchAll(/self\.__next_f\.push\(\[1,"((?:[^"\\]|\\.)*)"\]\)/g)].map((m) => m[1])
   264→  if (!chunks.length) return { metrics: {}, otherPeriods: [], parsed: false }
   265→  const flight = unescapeNextFlight(chunks.join(""))
   266→
   267→  // block boundaries: every ["$","div","<id>",{"className":"card|grid...
   268→  const marks = []
   269→  const markRe = /\["\$","div","([a-z_0-9]+)",\{"className":"(?:card|grid)/g
   270→  let mm
   271→  while ((mm = markRe.exec(flight))) marks.push({ id: mm[1], start: mm.index })
   272→  const metrics = {}
   273→  for (let i = 0; i < marks.length; i++) {
   274→    const segEnd = i + 1 < marks.length ? marks[i + 1].start : Math.min(marks[i].start + 4000, flight.length)
   275→    const seg = flight.slice(marks[i].start, segEnd)
   276→    const metric = ITEM_TO_METRIC[marks[i].id]
   277→    if (!metric) continue
   278→    const titles = [...seg.matchAll(/"title":"([0-9][0-9,.-]*)"/g)].map((m) => Number(m[1].replace(/,/g, "")))
   279→    const isHero = /className":"card/.test(seg.slice(0, 80))
   280→    const entry = metrics[metric] ?? {}
   281→    if (titles[0] !== undefined && Number.isFinite(titles[0])) entry.current = titles[0]
   282→    if (!isHero && titles[1] !== undefined && Number.isFinite(titles[1])) entry.prior = titles[1]
   283→    metrics[metric] = entry
   284→  }
   285→  return { metrics, otherPeriods: periodHrefPeriods(html, ticker), parsed: true }
   286→}

   289→function periodEnd(periodKey, year, startMonth) {
   290→  const sm = startMonth && startMonth >= 1 && startMonth <= 12 ? startMonth : 1
   291→  const startYear = sm === 1 ? year : year - 1
   292→  const monthAfter = sm - 1 + FY_END_MONTH_OFFSET[periodKey]
   293→  return new Date(Date.UTC(startYear, monthAfter, 0))
   294→}

   310→// ---------------------------------------------------------------- harvest
   311→async function harvest(tickers) {
   312→  log(`harvest: fetching ${tickers.length} overview pages`)
   313→  const overviews = await mapLimit(tickers, CONCURRENCY, async (ticker) => {
   314→    const url = `https://askborsa.com/company/${encodeURIComponent(ticker)}`
   315→    const res = await fetchWithRetry(url)
   316→    if (res.status === 404) return { ticker, status: "not_found" }
   317→    if (res.status !== 200) return { ticker, status: "fetch_failed", error: res.error ?? `HTTP ${res.status}` }
   318→    const meta = extractOverviewMeta(res.html)
   319→    const pivots = extractPivots(res.html)
   320→    const periods = periodHrefPeriods(res.html, ticker)
   321→    return { ticker, status: "ok", url, ...meta, pivots: pivots ?? [], overviewPeriods: periods }
   322→  })
   323→
   324→  // build statement-page fetch plan
   325→  const fetchPlan = new Map() // ticker -> Set(periodKey "q3-2026")
   326→  for (const o of overviews) {
   327→    if (o.status !== "ok") continue
   328→    const set = new Set()
   329→    const covered = new Set()
   330→    for (const p of o.pivots ?? []) {
   331→      for (const y of p.years ?? []) {
   332→        const key = `${p.period.toLowerCase()}-${y}`
   333→        covered.add(key)
   334→        // pivot-covered periods still get a page fetch when the pivot row set
   335→        // is thin (pivot lacks stock/cf rows for that period) — skip for now;
   336→        // comparatives from newer pages backfill older years anyway.
   337→      }
   338→    }
   339→    for (const key of o.overviewPeriods ?? []) if (!covered.has(key)) set.add(key)
   340→    // generated candidates: years beyond each pivot window up to CURRENT_YEAR
   341→    const maxByType = {}
   342→    for (const p of o.pivots ?? []) {
   343→      const t = p.period.toLowerCase()
   344→      for (const y of p.years ?? []) maxByType[t] = Math.max(maxByType[t] ?? 0, y)
   345→    }
   346→    for (const t of PERIOD_TYPES) {
   347→      // no pivot at all → probe a bounded recent window (2023+) and let BFS find older
   348→      const from = maxByType[t] ? maxByType[t] + 1 : 2023
   349→      for (let y = Math.max(from, 2019); y <= CURRENT_YEAR; y++) {
   350→        const key = `${t}-${y}`
   351→        if (!covered.has(key)) set.add(key)
   352→      }
   353→    }
   354→    fetchPlan.set(o.ticker, set)
   355→  }
   356→
   357→  let pages = [...fetchPlan.entries()].flatMap(([t, keys]) => [...keys].map((k) => ({ ticker: t, key: k })))
   358→  log(`statement pages to fetch (wave 1): ${pages.length}`)
   359→
   360→  const pageData = new Map() // `${ticker}|${key}` -> { metrics, otherPeriods }
   361→  async function fetchStatementWave(items) {
   362→    await mapLimit(items, CONCURRENCY, async ({ ticker, key }) => {
   363→      const id = `${ticker}|${key}`
   364→      const url = `https://askborsa.com/company/${encodeURIComponent(ticker)}/${key}`
   365→      const res = await fetchWithRetry(url)
   366→      if (res.status !== 200) {
   367→        pageData.set(id, { status: res.status === 404 ? "missing" : "fetch_failed", error: res.error })
   368→        return
   369→      }
   370→      const parsed = parseStatementPage(res.html, ticker)
   371→      pageData.set(id, { status: "ok", url, ...parsed })
   372→    })
   373→  }
   374→  await fetchStatementWave(pages)
   375→
   376→  // BFS wave 2: "فترات أخرى" links discovered on fetched pages, not yet covered
   377→  const wave2 = []
   378→  for (const [id, d] of pageData) {
   379→    if (d.status !== "ok") continue
   380→    const [ticker] = id.split("|")
   381→    const plan = fetchPlan.get(ticker) ?? new Set()
   382→    for (const key of d.otherPeriods ?? []) {
   383→      if (!plan.has(key) && !pageData.has(`${ticker}|${key}`)) {
   384→        plan.add(key)
   385→        wave2.push({ ticker, key })
   386→      }
   387→    }
   388→  }
   389→  if (wave2.length) {
   390→    log(`statement pages to fetch (wave 2, discovered links): ${wave2.length}`)
   391→    await fetchStatementWave(wave2)
   392→  }
   393→
   394→  // merge everything into a unified snapshot
   395→  const results = []
   396→  for (const o of overviews) {
   397→    const out = {
   398→      ticker: o.ticker,
   399→      status: o.status,
   400→      currency: o.currency ?? null,
   401→      fiscalYearStartMonth: o.fiscalYearStartMonth ?? null,
   402→      fiscalMonthAr: o.fiscalMonthAr ?? null,
   403→      sourceUrl: o.url ?? null,
   404→      periods: {}, // "q3-2026" -> { metric: {value, source} }
   405→      pivotCovered: [],
   406→    }
   407→    if (o.status === "ok") {
   408→      for (const p of o.pivots ?? []) {
   409→        const t = p.period.toLowerCase()
   410→        for (const y of p.years ?? []) {
   411→          const key = `${t}-${y}`
   412→          out.pivotCovered.push(key)
   413→          const bucket = out.periods[key] ??= {}
   414→          for (const row of p.rows ?? []) {
   415→            const metric = ITEM_TO_METRIC[row.item]
   416→            if (!metric) continue
   417→            const v = row.values?.[String(y)]
   418→            if (v === null || v === undefined || !Number.isFinite(Number(v)) || Number(v) === 0) continue
   419→            bucket[metric] = { value: Number(v), source: "pivot" }
   420→          }
   421→        }
   422→      }
   423→    }
   424→    for (const [id, d] of pageData) {
   425→      if (!id.startsWith(`${o.ticker}|`) || d.status !== "ok") continue
   426→      const key = id.split("|")[1]
   427→      const bucket = out.periods[key] ??= {}
   428→      for (const [metric, vals] of Object.entries(d.metrics ?? {})) {
   429→        if (vals.current !== undefined && Number.isFinite(vals.current) && vals.current !== 0) {
   430→          bucket[metric] = { value: vals.current, source: "statement" }
   431→        }
   432→      }
   433→      // prior-year comparative fills gaps only
   434→      const [t, yStr] = key.split("-")
   435→      const priorKey = `${t}-${Number(yStr) - 1}`
   436→      if (Number(yStr) - 1 >= 2015) {
   437→        const priorBucket = out.periods[priorKey] ??= {}
   438→        for (const [metric, vals] of Object.entries(d.metrics ?? {})) {
   439→          if (vals.prior === undefined || !Number.isFinite(vals.prior) || vals.prior === 0) continue
   440→          if (!priorBucket[metric]) priorBucket[metric] = { value: vals.prior, source: "comparative" }
   441→        }
   442→      }
   443→    }
   444→    results.push(out)
   445→  }
   446→
   447→  const summary = {}
   448→  for (const r of results) summary[r.status] = (summary[r.status] ?? 0) + 1
   449→  const periodsFetched = [...pageData.values()].filter((d) => d.status === "ok").length
   450→  log(`harvest done: companies=${JSON.stringify(summary)}, statementPagesOK=${periodsFetched}`)
   451→  return { generatedAt: new Date().toISOString(), currentYear: CURRENT_YEAR, summary, statementPagesFetched: periodsFetched, results }
   452→}

   454→// ---------------------------------------------------------------- pattern votes
   455→// Per (ticker, metric): vote "discrete" | "cumulative" | null from years where
   456→// the full flow series + FY value exist on AskBorsa.
   457→function votePatterns(snapshot) {
   458→  const votes = new Map() // ticker -> metric -> { discrete, cumulative, years: [] }
   459→  for (const r of snapshot.results) {
   460→    if (r.status !== "ok") continue
   461→    // collect per fiscal year flow series: values for q1..q4 + fy
   462→    const byYear = new Map()
   463→    for (const [key, bucket] of Object.entries(r.periods)) {
   464→      const [t, yStr] = key.split("-")
   465→      const y = Number(yStr)
   466→      for (const metric of FLOW) {
   467→        const v = bucket[metric]?.value
   468→        if (v === undefined || !Number.isFinite(v) || v === 0) continue
   469→        if (!byYear.has(y)) byYear.set(y, {})
   470→        const rec = byYear.get(y)
   471→        ;(rec[t] ??= {})[metric] = v
   472→      }
   473→    }
   474→    for (const [y, rec] of byYear) {
   475→      const fyVals = rec.fy ?? {}
   476→      for (const metric of FLOW) {
   477→        const fy = fyVals[metric]
   478→        if (fy === undefined) continue
   479→        const qs = ["q1", "q2", "q3", "q4"].map((t) => rec[t]?.[metric])
   480→        if (qs.some((v) => v === undefined)) continue // need full series
   481→        const [q1, q2, q3, q4] = qs
   482→        let vote = null
   483→        if (closeSum(q1 + q2 + q3 + q4, fy) && ![q1, q2, q3].some((v) => approx(v, fy)) && !([q1, q2, q3].every((v) => v === 0))) {
   484→          vote = "discrete"
   485→        } else {
   486→          // cumulative: same sign, |magnitudes| non-decreasing, terminal ≈ FY
   487→          const series = [q1, q2, q3, q4]
   488→          const signs = series.map(sameSign)
   489→          const sign = signs.find((s) => s !== 0)
   490→          const monotone = sign !== undefined && signs.every((s) => s === sign || s === 0)
   491→            && series.every((v, i) => i === 0 || Math.abs(v) >= Math.abs(series[i - 1]) * 0.995)
   492→          const terminalOk = approx(q4, fy) && Math.abs(q4 / fy - 1) <= 0.01
   493→          const accumulation = Math.abs(q3) >= Math.abs(q1) * 1.2
   494→          if (monotone && terminalOk && accumulation) vote = "cumulative"
   495→        }
   496→        if (vote) {
   497→          if (!votes.has(r.ticker)) votes.set(r.ticker, {})
   498→          const v = votes.get(r.ticker)
   499→          v[metric] ??= { discrete: 0, cumulative: 0, years: [] }
   500→          v[metric][vote]++
   501→          v[metric].years.push(`${y}:${vote}`)
   502→        }
   503→      }
   504→    }
   505→  }
   506→  const patterns = new Map()
   507→  for (const [ticker, byMetric] of votes) {
   508→    for (const [metric, v] of Object.entries(byMetric)) {
   509→      const winner = v.discrete === v.cumulative ? null : (v.discrete > v.cumulative ? "discrete" : "cumulative")
   510→      patterns.set(`${ticker}|${metric}`, { pattern: winner, discrete: v.discrete, cumulative: v.cumulative })
   511→    }
   512→  }
   513→  return patterns
   514→}

   690→          if (!(ratio >= 0.1 && ratio <= 10)) { skip(r.ticker, qRec.key, metric, `STOCK_SCALE_SUSPECT (${ratio.toFixed(2)}x)`); continue }

   705→            const fy = fyRec.bucket[metric]?.value

   707→                closeSum(qvals.reduce((a, b) => a + b, 0), fy) &&

   721→              const full = ["q1", "q2", "q3", "q4"].every((tt) => rec[tt]?.bucket[metric]?.value !== undefined)
   722→              if (full) skip(r.ticker, qRec.key, metric, "FLOW_GATE_FAILED_FULL_SERIES")
   723→            }
   724→          } else {
   725→            // current year: historical pattern + monotonicity + prior-year ratio
   726→            const pat = patterns.get(`${r.ticker}|${metric}`)
   727→            if (!pat?.pattern) { skip(r.ticker, qRec.key, metric, "FLOW_NO_PATTERN"); continue }
   728→            const priorVal = r.periods[`${t}-${y - 1}`]?.[metric]?.value
   729→            if (priorVal === undefined) { skip(r.ticker, qRec.key, metric, "FLOW_NO_PRIOR_YEAR"); continue }
   730→            const ratio = Math.abs(v) / Math.abs(priorVal)
   731→            if (!(ratio >= 0.05 && ratio <= 20)) { skip(r.ticker, qRec.key, metric, `FLOW_PRIOR_RATIO_SUSPECT (${ratio.toFixed(2)}x)`); continue }
   732→            const mono = cumulativeMonotoneUpTo(rec, metric, t)
   733→            if (mono === false) { skip(r.ticker, qRec.key, metric, "FLOW_MONOTONICITY_BROKEN"); continue }
   734→            if (pat.pattern === "discrete") {
   735→              target = { pt: "QUARTERLY", label: `${t.toUpperCase()} ${y}` }
   736→              verifiedBy = "PATTERN_DISCRETE"
   737→            } else if (pat.pattern === "cumulative") {
   738→              if (t === "q4") { skip(r.ticker, qRec.key, metric, "CUMULATIVE_Q4_COVERED_BY_FY"); continue }
   739→              const tgt = { q1: ["QUARTERLY", `Q1 ${y}`], q2: ["SEMIANNUAL", `H1 ${y}`], q3: ["NINE_MONTH", `9M ${y}`] }[t]
   740→              target = { pt: tgt[0], label: tgt[1] }
   741→              verifiedBy = "PATTERN_CUMULATIVE"
   742→            }
   743→          }
   744→          if (!target) continue
   745→          const tk = `${target.pt}|${target.label}`
   746→          const arr = flowByTarget.get(tk) ?? []
   747→          arr.push({ metric, value: v, verifiedBy })
   748→          flowByTarget.set(tk, arr)
   749→        }

   751→        // emit plans for this quarter
   752→        const targets = new Map()
   753→        if (stockValues.length) targets.set(`QUARTERLY|${t.toUpperCase()} ${y}`, { pt: "QUARTERLY", label: `${t.toUpperCase()} ${y}`, values: stockValues })
   754→        for (const [tk, vals] of flowByTarget) {
   755→          const [pt, label] = tk.split("|")
   756→          const ex = targets.get(tk)
   757→          if (ex) ex.values.push(...vals)
   758→          else targets.set(tk, { pt, label, values: vals })
   759→        }
   760→        for (const [tk, spec] of targets) {
   761→          const existingReport = per.get(reportKey(spec.pt, spec.label))
   762→          // per-value sanity vs the app's own annuals of the same metric.
   763→          // flows get a wider band than stocks: seasonal quarters can be a
   764→          // tiny fraction of the year (GPIM Q1 ≈ 4% of FY), while a balance
   765→          // sheet cannot plausibly collapse to 5% year-over-year. Real unit
   766→          // errors are ~1000x and are blocked by both bands.
   767→          const saneValues = []
   768→          for (const v of spec.values) {
   769→            const ref = nearestAnnual(annualRef.get(v.metric), y)
   770→            if (ref === undefined) { saneValues.push(v); continue }
   771→            const ratio = Math.abs(v.value) / Math.abs(ref)
   772→            const [lo, hi] = FLOW.has(v.metric) ? [0.01, 50] : [0.05, 20]
   773→            if (ratio >= lo && ratio <= hi) saneValues.push(v)
   774→            else skip(r.ticker, qRec.key, v.metric, `VALUE_SCALE_VS_APP (${ratio.toExponential(1)}x vs ${v.metric} annual)`)
   775→          }
   776→          if (!saneValues.length) continue
   777→          spec.values = saneValues
   778→          // a Q4 report carrying ONLY point-in-time stock values would just
   779→          // duplicate the annual balance sheet — never create it
   780→          if (!existingReport && t === "q4" && !spec.values.some((v) => FLOW.has(v.metric))) {
   781→            skip(r.ticker, qRec.key, null, "Q4_STOCK_ONLY_NO_REPORT")
   782→            continue
   783→          }
   784→          if (!existingReport) {
   785→            plan.push({
   786→              action: "CREATE_REPORT", kind: "QUARTER", ticker: r.ticker, companyId: company.id,
   787→              periodType: spec.pt, periodLabel: spec.label, fiscalYear: y,
   788→              periodStart: quarterStart(end).toISOString(), periodEnd: end.toISOString(),
   789→              sourceUrl: r.sourceUrl, confidence: spec.values.some((v) => String(v.verifiedBy).startsWith("PATTERN")) ? 0.7 : 0.8,
   790→              verifiedBy: spec.values.map((v) => v.verifiedBy).join("+"),
   791→              values: spec.values,
   792→            })
   793→          } else {
   794→            const have = new Set(existingReport.values
   795→              .filter((v) => v.subPeriod === "CURRENT" || v.subPeriod === null)
   796→              .map((v) => v.metricCode))
   797→            const fills = spec.values.filter((v) => !have.has(v.metric))
   798→            if (fills.length) {
   799→              plan.push({
   800→                action: "ADD_VALUE", kind: "QUARTER", ticker: r.ticker, companyId: company.id,
   801→                reportId: existingReport.id, reportKey: spec.label, fiscalYear: y,
   802→                sourceUrl: r.sourceUrl, confidence: fills.some((v) => String(v.verifiedBy).startsWith("PATTERN")) ? 0.7 : 0.8,
   803→                verifiedBy: fills.map((v) => v.verifiedBy).join("+"),
   804→                values: fills,
   805→              })
   806→            }
   807→            for (const v of spec.values) {
   808→              const exV = existingReport.values.find((x) => x.metricCode === v.metric && (x.subPeriod === "CURRENT" || x.subPeriod === null) && Number.isFinite(x.normalizedValue))
   809→              if (exV) {
   810→                const d = relDiff(v.value, exV.normalizedValue)
   811→                if (d !== null && d > 0.05) conflicts.push({ ticker: r.ticker, period: spec.label, metric: v.metric, askValue: v.value, appValue: exV.normalizedValue, relDiff: Number(d.toFixed(4)), askSource: "askborsa" })
   812→              }
   813→            }
   814→          }
   815→        }
   816→      }
   817→    }
   818→  }
   819→
   820→  return { plan, skipped, conflicts }
   821→}

   823→function nearestAnnual(arr, year) {
   824→  if (!arr?.length) return undefined
   825→  let best
   826→  for (const a of arr) {
   827→    const dist = Math.abs(a.year - year)
   828→    if (!best || dist < best.dist) best = { dist, value: a.value }
   829→  }
   830→  return best.value
   831→}

   833→// cumulative check for a year with ask FY: series Q1..Q3 (+Q4 or FY as terminal)
   834→function cumulativeCheck(rec, metric, fy) {
   835→  const vals = ["q1", "q2", "q3"].map((t) => rec[t]?.bucket[metric]?.value)
   836→  if (vals.some((v) => v === undefined || v === 0)) return false
   837→  const q4 = rec.q4?.bucket[metric]?.value
   838→  const series = q4 !== undefined && q4 !== 0 ? [...vals, q4] : [...vals, fy]
   839→  if (series.some((v) => v === undefined || v === 0)) return false
   840→  const sign = sameSign(series[0])
   841→  if (sign === 0) return false
   842→  const monotone = series.every((v) => sameSign(v) === sign) && series.every((v, i) => i === 0 || Math.abs(v) >= Math.abs(series[i - 1]) * 0.995)
   843→  const terminalOk = q4 !== undefined && q4 !== 0 ? approx(q4, fy) && Math.abs(q4 / fy - 1) <= 0.01 : true
   844→  const accumulation = Math.abs(vals[2]) >= Math.abs(vals[0]) * 1.2
   845→  // must NOT pass the discrete sum gate (that case is handled first)
   846→  const allFour = ["q1", "q2", "q3", "q4"].map((t) => rec[t]?.bucket[metric]?.value)
   847→  const discretePass = allFour.every((v) => v !== undefined) && closeSum(allFour.reduce((a, b) => a + b, 0), fy)
   848→  return monotone && terminalOk && accumulation && !discretePass
   849→}

   851→// monotonicity of cumulative pattern up to quarter t (for current-year import)
   852→function cumulativeMonotoneUpTo(rec, metric, upTo) {
   853→  const idx = { q1: 1, q2: 2, q3: 3, q4: 4 }[upTo]
   854→  const series = []
   855→  for (const t of ["q1", "q2", "q3", "q4"].slice(0, idx)) {
   856→    const v = rec[t]?.bucket[metric]?.value
   857→    if (v !== undefined && v !== 0) series.push(v)
   858→  }
   859→  if (series.length < 2) return true // nothing to contradict
   860→  const sign = sameSign(series[0])
   861→  return sign !== 0 && series.every((v) => sameSign(v) === sign) && series.every((v, i) => i === 0 || Math.abs(v) >= Math.abs(series[i - 1]) * 0.995)
   862→}

   865→async function applyPlan(plan) {
   866→  mkdirSync("reports", { recursive: true })
   867→  const rollback = { generatedAt: new Date().toISOString(), reports: [], values: [] }
   868→  let reportCount = 0
   869→  let valueCount = 0
   870→  for (const p of plan) {
   871→    let reportId
   872→    if (p.action === "CREATE_REPORT") {
   873→      const dup = await db.financialReport.findFirst({
   874→        where: { companyId: p.companyId, periodType: p.periodType, periodLabel: p.periodLabel, statementType: "CONSOLIDATED", deletedAt: null, isDemoData: false },
   875→        select: { id: true },
   876→      })
   877→      if (dup) {
   878→        reportId = dup.id
   879→      } else {
   880→        const created = await db.financialReport.create({
   881→          data: {
   882→            companyId: p.companyId,
   883→            reportType: p.periodType === "ANNUAL" ? "ANNUAL" : "INTERIM",
   884→            periodType: p.periodType,
   885→            statementType: "CONSOLIDATED",
   886→            fiscalYear: p.fiscalYear,
   887→            periodLabel: p.periodLabel,
   888→            periodStart: new Date(p.periodStart),
   889→            periodEnd: new Date(p.periodEnd),
   890→            language: "AR",
   891→            sourceName: "ASKBORA_PUBLIC",
   892→            sourceUrl: p.sourceUrl,
   893→            processingStatus: "EXTRACTED",
   894→            extractionMethod: "ASKBORA_HARVEST",
   895→            parserVersion: PARSER,
   896→            extractionConfidence: p.confidence,
   897→            isDemoData: false,
   898→            version: 1,
   899→            notes: `AskBorsa public data sync; verified: ${p.verifiedBy}.`,
   900→          },
   901→          select: { id: true },
   902→        })
   903→        reportId = created.id
   904→        rollback.reports.push({ id: reportId, ticker: p.ticker, periodLabel: p.periodLabel })
   905→        reportCount++
   906→      }
   907→    } else {
   908→      reportId = p.reportId
   909→    }
   910→    for (const v of p.values) {
   911→      // final only-missing-slot re-check under concurrency-safe conditions
   912→      const slot = await db.financialValue.findFirst({
   913→        where: { reportId, metricCode: v.metric, OR: [{ subPeriod: "CURRENT" }, { subPeriod: null }] },
   914→        select: { id: true },
   915→      })
   916→      if (slot) continue
   917→      const row = await db.financialValue.create({
   918→        data: {
   919→          companyId: p.companyId,
   920→          reportId,
   921→          metricCode: v.metric,
   922→          // Product rule: originalLabel is user-visible (bilingual hint on the
   923→          // statements table) — never embed tooling/source refs like
   924→          // "(AskBorsa Qx)" here. Provenance lives in validationNotes instead.
   925→          originalLabel: AR_LABEL[v.metric] ?? EN_FALLBACK[v.metric] ?? prettyCode(v.metric),
   926→          value: v.value,
   927→          currency: "EGP",
   928→          unit: "UNIT",
   929→          normalizedValue: v.value,
   930→          statementType: statementKind(v.metric),
   931→          subPeriod: "CURRENT",
   932→          extractionMethod: "ASKBORA_HARVEST",
   933→          confidence: p.confidence ?? 0.8,
   934→          validationStatus: "VALID",
   935→          validationNotes: `${MARK} ${v.verifiedBy}; source ${p.sourceUrl}`,
   936→          sourceText: p.sourceUrl,
   937→          isDemoData: false,
   938→        },
   939→        select: { id: true },
   940→      })
   941→      rollback.values.push({ id: row.id, ticker: p.ticker, metric: v.metric, value: v.value })
   942→      valueCount++
   943→    }
   944→  }
   945→  writeFileSync(ROLLBACK, JSON.stringify(rollback, null, 2), "utf8")
   946→  await db.auditLog.create({
   947→    data: {
   948→      actor: "system",
   949→      action: "UPLOAD",
   950→      entityType: "FinancialReport",
   951→      details: `ASKBORA_SYNC: ${reportCount} reports created, ${valueCount} values added (parser ${PARSER}); rollback ${ROLLBACK}`,
   952→    },
   953→  })
   954→  return { reportCount, valueCount }
   955→}

   957→async function runRollback() {
   958→  if (!existsSync(ROLLBACK)) { console.error(`No rollback file at ${ROLLBACK}`); process.exit(1) }
   959→  const data = JSON.parse(readFileSync(ROLLBACK, "utf8"))
   960→  let values = 0
   961→  let reports = 0
   962→  let skipped = 0
   963→  for (const v of data.values ?? []) {
   964→    const row = await db.financialValue.findUnique({ where: { id: v.id }, select: { validationNotes: true } })
   965→    if (!row || !String(row.validationNotes ?? "").startsWith(MARK)) { skipped++; continue }
   966→    await db.financialValue.delete({ where: { id: v.id } })
   967→    values++
   968→  }
   969→  for (const r of data.reports ?? []) {
   970→    const row = await db.financialReport.findUnique({ where: { id: r.id }, select: { sourceName: true, _count: { select: { values: true } } } })
   971→    if (!row || row.sourceName !== "ASKBORA_PUBLIC" || row._count.values > 0) { skipped++; continue }
   972→    await db.financialReport.delete({ where: { id: r.id } })
   973→    reports++
   974→  }
   975→  console.log(`rollback complete: values=${values}, reports=${reports}, skipped=${skipped}`)
   976→}
