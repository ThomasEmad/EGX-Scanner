   126→              <CardTitle className="text-sm">{t("sc.presets")}</CardTitle>

   190→                <Skeleton key={i} className="h-16 rounded-lg" />

   192→            </div>
   193→          ) : result ? (
   194→            <>
   195→              <div className="flex flex-wrap items-center gap-2 text-sm">
   196→                <span className="font-semibold">{result.ruleName}</span>
   197→                <Badge variant="outline">{result.periodBasis === "LATEST_ANNUAL" ? t("sc.basisAnnual") : t("sc.basisQuarterly")}</Badge>
   198→                <Badge className="tabular">
   199→                  {result.matched.length} / {result.results.length} {t("sc.matchingCompanies")}
   200→                </Badge>
   201→              </div>
   202→
   203→              {result.validationErrors.length > 0 ? (
   204→                <div className="flex items-start gap-2 rounded-lg border border-amber-500/40 bg-amber-500/10 p-3 text-xs text-amber-700 dark:text-amber-400">

   210→                        <li key={i}>{e}</li>

   212→                    </ul>

   248→        </div>

   275→            <div className="min-w-0">

   277→              <p className="flex items-center gap-1 text-[10px] text-muted-foreground">
   278→                <Clock className="h-2.5 w-2.5" />

   280→              </p>
   281→            </div>
   282→            <Badge variant="secondary" className="shrink-0 tabular text-[10px]">
   283→              {r.matchedCount} {t("rh.matches")}
   284→            </Badge>
   285→          </div>
   286→        ))}
   287→      </CardContent>
   288→    </Card>
   289→  )
   290→}
   291→
   292→function timeAgo(iso: string, lang: string): string {
   293→  const diff = Date.now() - new Date(iso).getTime()
   294→  const mins = Math.floor(diff / 60000)
   295→  if (mins < 1) return lang === "ar" ? "الآن" : "just now"
   296→  if (mins < 60) return lang === "ar" ? `قبل ${mins} د` : `${mins}m ago`
   297→  const hours = Math.floor(mins / 60)
   298→  if (hours < 24) return lang === "ar" ? `قبل ${hours} س` : `${hours}h ago`
   299→  return lang === "ar" ? `قبل ${Math.floor(hours / 24)} ي` : `${Math.floor(hours / 24)}d ago`
   300→}

   436→            {lang === "ar"
   437→              ? "اجمع شروطاً متعددة (منطق AND) من مؤشرات وأحداث وتوزيعات. كل نتيجة تُشرح من القيم المخزنة فعلياً."
   438→              : "Combine multiple conditions (AND logic) from metrics, events and dividends. Every result is explained from actual stored values."}
   439→          </p>
   440→        </div>
   441→        <Select value={basis} onValueChange={(v) => setBasis(v as typeof basis)}>
   442→          <SelectTrigger className="w-[230px] h-9">
   443→            <SelectValue />
   444→          </SelectTrigger>
   445→          <SelectContent>
   446→            <SelectItem value="LATEST_ANNUAL">{t("sc.basisAnnual")}</SelectItem>

   511→                    value={row.statusAny ? "active" : "paid"}

   588→        {/* results */}

   599→                <div className="rounded-lg border border-amber-500/40 bg-amber-500/10 p-3 text-xs text-amber-700 dark:text-amber-400">
   600→                  <ul className="list-disc ps-4 space-y-0.5">
   601→                    {result.validationErrors.map((e, i) => (
   602→                      <li key={i}>{e}</li>
   603→                    ))}
   604→                  </ul>
   605→                </div>
   606→              ) : null}
   607→              <div className="max-h-[560px] space-y-2 overflow-y-auto scrollbar-thin pe-1">
   608→                {result.results.map((r) => (

   633→              {t("sc.noResults")}
