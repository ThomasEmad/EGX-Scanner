    22→    22→//   • only-missing-slot: existing values are NEVER overwritten; conflicts (>5%)

   189→   189→function parseMonthYear(s) {
   190→   190→  // "Jun 2006" | "Dec 2025" → Date at month end (UTC)
   191→   191→  const m = String(s || "").trim().match(/^([A-Za-z]{3})\s+(\d{4})$/)
   192→   192→  if (!m) return null
   193→   193→  const mi = MONTHS[m[1].toLowerCase()]
   194→   194→  if (mi === undefined) return null
   195→   195→  const y = Number(m[2])
   196→   196→  return new Date(Date.UTC(y, mi + 1, 0, 23, 59, 59))
   197→   197→}

   201→   201→// Classify one column of a statement view. Returns a period object or null.
   202→   202→// tvFiscalYear = the fiscal year TradingView labels the column with (Q4 '18 →
   203→   203→// FY2018 even when the quarter ends Jun-2018) — used for the quarter-sum
   204→   204→// checks; the platform report itself keeps the calendar year of periodEnd.
   205→   205→function classifyColumn(header, subheader) {
   206→   206→  const end = parseMonthYear(subheader)
   207→   207→  if (!end) return null
   208→   208→  const now = Date.now() + 5 * 86_400_000
   209→   209→  if (end.getTime() > now) return { skip: "FUTURE" }
   210→   210→  const h = String(header || "").trim()
   211→   211→  const year = end.getUTCFullYear()
   212→   212→  const month = end.getUTCMonth()
   213→   213→  // "H1 '25" / "H2 '25" — semiannual columns
   214→   214→  let m = h.match(/^H([12])\s*'(\d{2})$/)
   215→   215→  if (m) {
   216→   216→    const half = Number(m[1])
   217→   217→    return { periodType: "SEMIANNUAL", periodLabel: `H${half} ${year}`, fiscalYear: year, tvFiscalYear: 2000 + Number(m[2]), periodEnd: end, kind: half === 1 ? "H1" : "H2", months: 6 }
   218→   218→  }
   219→   219→  // "Q3 '26" — quarterly columns (fiscal labels; calendar mapping comes from the end month)
   220→   220→  m = h.match(/^Q([1-4])\s*'(\d{2})$/)
   221→   221→  if (m) {
   222→   222→    const q = month <= 2 ? 1 : month <= 5 ? 2 : month <= 8 ? 3 : 4
   223→   223→    return { periodType: "QUARTERLY", periodLabel: `Q${q} ${year}`, fiscalYear: year, tvFiscalYear: 2000 + Number(m[2]), periodEnd: end, kind: `Q${q}`, months: 3 }
   224→   224→  }
   225→   225→  // "H1 2025" / "H2 2025" unquoted variant
   226→   226→  m = h.match(/^H([12])\s+(\d{4})$/)
   227→   227→  if (m) {
   228→   228→    const half = Number(m[1])
   229→   229→    return { periodType: "SEMIANNUAL", periodLabel: `H${half} ${year}`, fiscalYear: year, tvFiscalYear: Number(m[2]), periodEnd: end, kind: half === 1 ? "H1" : "H2", months: 6 }
   230→   230→  }
   231→   231→  // plain year "2006" — annual column
   232→   232→  if (/^\d{4}$/.test(h)) {
   233→   233→    return { periodType: "ANNUAL", periodLabel: `FY ${h}`, fiscalYear: Number(h), tvFiscalYear: Number(h), periodEnd: end, kind: "FY", months: 12 }
   234→   234→  }
   235→   235→  return null
   236→   236→}
   237→   237→function classifyView(headers, subheaders) {
   238→   238→  const out = []
   239→   239→  const seen = new Map()
   240→   240→  let restated = 0
   241→   241→  for (let i = 0; i < headers.length; i++) {
   242→   242→    const p = classifyColumn(headers[i], subheaders[i])
   243→   243→    if (!p) { out.push(null); continue }
   244→   244→    if (p.skip) { out.push(null); continue }
   245→   245→    const key = `${p.periodType}|${p.periodLabel}`
   246→   246→    if (seen.has(key)) {
   247→   247→      // same label appearing twice: keep the later column (restatement), log it
   248→   248→      const prevIdx = seen.get(key)
   249→   249→      out[prevIdx] = null
   250→   250→      restated++
   251→   251→      seen.set(key, i)
   252→   252→      out.push(p)
   253→   253→      continue
   254→   254→    }
   255→   255→    seen.set(key, i)
   256→   256→    out.push(p)
   257→   257→  }
   258→   258→  return { periods: out, restated }
   259→   259→}

   261→   261→// ------------------------------------------------------------ harvest ----
   262→   262→// Runs inside the page. Walks the React fiber of every table row to recover
   263→   263→// the EXACT values (the DOM shows rounded "8.84 B"; the fiber holds the raw
   264→   264→// numbers the WebSocket payload delivered).
   265→   265→const EXTRACT_FN = `
   266→   266→(() => {
   267→   267→  const rows = [];
   268→   268→  const seen = new Set();
   269→   269→  for (const el of document.querySelectorAll('[data-name]')) {
   270→   270→    let hit = null;
   271→   271→    for (const k of Object.keys(el)) {
   272→   272→      if (!k.startsWith('__reactFiber')) continue;
   273→   273→      let cur = el[k], d = 0;
   274→   274→      while (cur && d < 6) {
   275→   275→        const p = cur.memoizedProps;
   276→   276→        if (p && p.idWithoutPeriod && Array.isArray(p.values) && p.values.length > 0) { hit = p; break; }
   277→   277→        cur = cur.return; d++;
   278→   278→      }
   279→   279→      if (hit) break;
   280→   280→    }
   281→   281→    if (!hit || seen.has(hit.idWithoutPeriod)) continue;
   282→   282→    seen.add(hit.idWithoutPeriod);
   283→   283→    rows.push({ id: hit.idWithoutPeriod, title: el.getAttribute('data-name'), values: hit.values, additional: typeof hit.additionalValue === 'number' ? hit.additionalValue : null });
   284→   284→  }
   285→   285→  let headers = null, subheaders = null;
   286→   286→  // walk from every data-name element until one reaches the table component
   287→   287→  // (nav elements never do — the component is an ancestor of the ROWS)
   288→   288→  const els = document.querySelectorAll('[data-name]');
   289→   289→  outer: for (const el of els) {
   290→   290→    for (const k of Object.keys(el)) {
   291→   291→      if (!k.startsWith('__reactFiber')) continue;
   292→   292→      let cur = el[k], d = 0;
   293→   293→      while (cur && d < 45) {
   294→   294→        const p = cur.memoizedProps;

   296→   296→          headers = p.headerValues.map((x) => String(x));
   297→   297→          subheaders = Array.isArray(p.headerSubvalues) ? p.headerSubvalues.map((x) => String(x)) : null;
   298→   298→          break outer;
   299→   299→        }
   300→   300→        cur = cur.return; d++;
   301→   301→      }
   302→   302→    }
   303→   303→  }
   304→   304→  // page metadata: symbol, currency, report frequency (from SSR init-data)
   305→   305→  let symbol = null, currency = null, frequency = null, fiscalQ = null, fiscalFY = null;
   306→   306→  for (const s of document.querySelectorAll('script[type="application/prs.init-data+json"]')) {
   307→   307→    try {
   308→   308→      const j = JSON.parse(s.textContent);
   309→   309→      for (const v of Object.values(j)) {
   310→   310→        if (v && typeof v === 'object') {
   311→   311→          if (typeof v.symbol === 'string' && !symbol) symbol = v.symbol;
   312→   312→          const q = v.initFinancialsQuotes;
   313→   313→          if (q) {
   314→   314→            currency = currency || (q.fundamental_currency_code || '').trim();
   315→   315→            frequency = frequency || q.last_report_frequency;
   316→   316→            fiscalQ = Array.isArray(q.fiscal_period_fq_h) ? q.fiscal_period_fq_h : null;
   317→   317→            fiscalFY = Array.isArray(q.fiscal_period_fy_h) ? q.fiscal_period_fy_h : null;
   318→   318→          }
   319→   319→        }
   320→   320→      }
   321→   321→    } catch {}
   322→   322→  }
   323→   323→  return { rows, headers, subheaders, symbol, currency, frequency, fiscalQ, fiscalFY, url: location.href };
   324→   324→})()

   334→   334→// Light probe used by waitForFunction: counts fiber-backed table rows and
   335→   335→// whether the current column labels are quarterly-style. The financials table
   336→   336→// mounts asynchronously (data arrives over a WebSocket), so the page must be
   337→   337→// polled — a fixed sleep is not reliable.
   338→   338→const PROBE_FN = `
   339→   339→(() => {
   340→   340→  let n = 0, quarterly = false, headers = null;
   341→   341→  for (const el of document.querySelectorAll('[data-name]')) {
   342→   342→    for (const k of Object.keys(el)) {
   343→   343→      if (!k.startsWith('__reactFiber')) continue;
   344→   344→      let cur = el[k], d = 0;
   345→   345→      while (cur && d < 6) {
   346→   346→        const p = cur.memoizedProps;
   347→   347→        if (p && p.idWithoutPeriod && Array.isArray(p.values) && p.values.length > 0) { n++; break; }
   348→   348→        cur = cur.return; d++;
   349→   349→      }
   350→   350→    }
   351→   351→  }
   352→   352→  // the headerValues-holding component is an ancestor of the financial rows —
   353→   353→  // nav/[data-name] elements never reach it, so walk from EVERY data-name
   354→   354→  // element until one succeeds (the first table row wins)
   355→   355→  const els = document.querySelectorAll('[data-name]');
   356→   356→  outer: for (const el of els) {
   357→   357→    for (const k of Object.keys(el)) {
   358→   358→      if (!k.startsWith('__reactFiber')) continue;
   359→   359→      let cur = el[k], d = 0;
   360→   360→      while (cur && d < 45) {
   361→   361→        const p = cur.memoizedProps;
   362→   362→        if (p && Array.isArray(p.headerValues) && p.headerValues.length > 0 && Array.isArray(p.items)) {
   363→   363→          headers = p.headerValues.map(String);
   364→   364→          break outer;
   365→   365→        }
   366→   366→        cur = cur.return; d++;
   367→   367→      }
   368→   368→    }
   369→   369→  }
   370→   370→  if (headers) quarterly = headers.some((h) => /^Q[1-4]\\s*'/.test(h) || /^H[12]\\s*'/.test(h));
   371→   371→  return { n, quarterly };
   372→   372→})()
   373→   373→`

   375→   375→async function extractView(page, view) {
   376→   376→  if (view === "quarterly") {
   377→   377→    const btn = page.locator("button", { hasText: /^Quarterly$/ }).first()
   378→   378→    try {
   379→   379→      await btn.click({ timeout: 5_000 })
   380→   380→    } catch {
   381→   381→      return { error: "QUARTERLY_TOGGLE_MISSING" }
   382→   382→    }
   383→   383→  }
   384→   384→  // poll until the table the view needs is actually mounted
   385→   385→  const wantQuarterly = view === "quarterly"
   386→   386→  const deadline = Date.now() + 25_000
   387→   387→  let probe = { n: 0, quarterly: false }
   388→   388→  let lastProbeErr = null
   389→   389→  const t0v = Date.now()
   390→   390→  while (Date.now() < deadline) {
   391→   391→    probe = await page.evaluate(PROBE_FN).catch((e) => { lastProbeErr = e.message.slice(0, 100); return { n: 0, quarterly: false } })
   392→   392→    if (process.env.TV_DEBUG) log(`  [probe ${view}] t+${Date.now() - t0v}ms n=${probe.n} q=${probe.quarterly} err=${lastProbeErr ?? "-"}`)
   393→   393→    if (probe.n > 0 && (!wantQuarterly || probe.quarterly)) break
   394→   394→    if (view === "quarterly" && probe.n > 0 && !probe.quarterly) {
   395→   395→      // annual table still showing — the click may have raced the render; retry
   396→   396→      await page.locator("button", { hasText: /^Quarterly$/ }).first().click({ timeout: 3_000 }).catch(() => {})
   397→   397→    }
   398→   398→    await sleep(700)
   399→   399→  }
   400→   400→  if (probe.n === 0) return { error: `TABLE_NOT_FOUND${lastProbeErr ? " (probe: " + lastProbeErr + ")" : ""}` }
   401→   401→  if (wantQuarterly && !probe.quarterly) return { error: "QUARTERLY_TOGGLE_FAILED" }
   402→   402→  await sleep(500) // let value arrays settle after mount
   403→   403→  const raw = await page.evaluate(EXTRACT_FN).catch((e) => ({ error: "EVAL_FAILED:" + e.message.slice(0, 120) }))
   404→   404→  if (raw.error) return raw
   405→   405→  if (!raw.headers || raw.rows.length === 0) return { error: "TABLE_NOT_FOUND" }
   406→   406→  return raw
   407→   407→}

   409→   409→async function harvestStatement(browser, url, statement) {
   410→   410→  const context = await browser.newContext({ viewport: { width: 1600, height: 900 }, locale: "en-US" })
   411→   411→  const page = await context.newPage()
   412→   412→  const out = { url, annual: null, quarterly: null }
   413→   413→  try {
   414→   414→    const res = await page.goto(url, { waitUntil: "domcontentloaded", timeout: PAGE_LOAD_TIMEOUT })
   415→   415→    if (res && (res.status() === 404 || res.status() === 410)) {
   416→   416→      out.notFound = true
   417→   417→      return out
   418→   418→    }
   419→   419→    if (res && res.status() >= 400) {
   420→   420→      out.error = `HTTP_${res.status()}`
   421→   421→      return out
   422→   422→    }
   423→   423→    await page.waitForSelector("[data-name]", { timeout: 20_000 }).catch(() => {})
   424→   424→    out.annual = await extractView(page, "annual")
   425→   425→    out.quarterly = await extractView(page, "quarterly")
   426→   426→  } catch (e) {
   427→   427→    out.error = e.message.slice(0, 200)
   428→   428→  } finally {
   429→   429→    await context.close().catch(() => {})
   430→   430→  }
   431→   431→  return out
   432→   432→}

   453→   453→async function harvestCompany(browser, comp) {
   454→   454→  const t0 = Date.now()
   455→   455→  const result = {
   456→   456→    ticker: comp.ticker,
   457→   457→    nameEn: comp.nameEn,
   458→   458→    currency: null,
   459→   459→    frequency: null,
   460→   460→    tvSymbol: null,
   461→   461→    notFound: false,
   462→   462→    errors: [],
   463→   463→    statements: {},
   464→   464→    fetchedAt: new Date().toISOString(),
   465→   465→  }
   466→   466→  for (const st of STATEMENTS) {
   467→   467→    let url = statementUrl(`EGX:${comp.ticker}`, st)
   468→   468→    let got = await harvestStatement(browser, url, st)
   469→   469→    if (got.notFound) {
   470→   470→      // try the original ticker (pre-rename), then the public symbol search
   471→   471→      const candidates = []
   472→   472→      if (comp.originalTicker && comp.originalTicker.toUpperCase() !== comp.ticker) candidates.push(`EGX:${comp.originalTicker.toUpperCase()}`)
   473→   473→      const search = await searchSymbol(comp.ticker)
   474→   474→      if (search?.length) candidates.push(...search.map((s) => s.symbol))
   475→   475→      let resolved = null
   476→   476→      for (const sym of candidates) {
   477→   477→        const u = statementUrl(sym, st)
   478→   478→        const retry = await harvestStatement(browser, u, st)
   479→   479→        if (!retry.notFound && !retry.error && retry.annual?.rows?.length) { resolved = { sym, url: u, got: retry }; break }
   480→   480→      }
   481→   481→      if (!resolved) { result.notFound = true; result.errors.push(`${st}:NO_SYMBOL_ON_TV`); continue }
   482→   482→      url = resolved.url
   483→   483→      got = resolved.got
   484→   484→    } else if (got.error) {
   485→   485→      result.errors.push(`${st}:${got.error}`)
   486→   486→      continue
   487→   487→    }
   488→   488→    result.tvSymbol = result.tvSymbol || got.annual?.symbol || `EGX:${comp.ticker}`
   489→   489→    result.currency = result.currency || got.annual?.currency || null
   490→   490→    result.frequency = result.frequency || got.annual?.frequency || null
   491→   491→    result.statements[st] = { url, annual: got.annual, quarterly: got.quarterly }
   492→   492→    await sleep(jitter(PAGE_DELAY_MS))
   493→   493→  }
   494→   494→  result.ms = Date.now() - t0
   495→   495→  return result
   496→   496→}

   498→   498→async function runHarvest(companies) {
   499→   499→  const { chromium: _c } = await import("playwright")
   500→   500→  const browser = await _c.launch({ headless: true, args: ["--no-sandbox", "--disable-dev-shm-usage"] })
   501→   501→  mkdirSync("reports", { recursive: true })
   502→   502→  const snapshot = existsSync(SNAPSHOT) ? JSON.parse(readFileSync(SNAPSHOT, "utf8")) : { generatedAt: null, companies: {} }
   503→   503→  let done = 0
   504→   504→  const errs = []
   505→   505→  let consecutiveFailures = 0
   506→   506→
   507→   507→  async function worker(queue) {
   508→   508→    while (queue.length) {
   509→   509→      if (BUDGET_MS && Date.now() - t0 > BUDGET_MS) { queue.length = 0; break } // time budget — resume on next run
   510→   510→      const comp = queue.shift()
   511→   511→      if (!comp) break
   512→   512→      let rec = null
   513→   513→      try {
   514→   514→        rec = await harvestCompany(browser, comp)
   515→   515→        consecutiveFailures = 0
   516→   516→      } catch (e) {
   517→   517→        rec = { ticker: comp.ticker, nameEn: comp.nameEn, error: "HARVEST_CRASH:" + e.message.slice(0, 160) }
   518→   518→        consecutiveFailures++
   519→   519→        if (consecutiveFailures >= 4) { log("circuit breaker: 4 consecutive crashes — aborting the run"); queue.length = 0 }
   520→   520→      }
   521→   521→      if (rec.error) errs.push({ ticker: rec.ticker, error: rec.error })
   522→   522→      else if (rec.errors?.length) errs.push({ ticker: rec.ticker, errors: rec.errors })
   523→   523→      snapshot.companies[rec.ticker] = rec
   524→   524→      snapshot.generatedAt = new Date().toISOString()
   525→   525→      done++
   526→   526→      if (done % 5 === 0 || done === companies.length) {
   527→   527→        writeFileSync(SNAPSHOT, JSON.stringify(snapshot), "utf8")
   528→   528→        const ok = Object.values(snapshot.companies).filter((c) => Object.keys(c.statements ?? {}).length > 0).length
   529→   529→        log(`harvest ${done}/${companies.length} (usable: ${ok}, last: ${rec.ticker})`)
   530→   530→      }
   531→   531→    }
   532→   532→  }
   533→   533→
   534→   534→  const queue = [...companies]
   535→   535→  const t0 = Date.now()
   536→   536→  await Promise.all(Array.from({ length: WORKERS }, () => worker(queue)))
   537→   537→  writeFileSync(SNAPSHOT, JSON.stringify(snapshot), "utf8")
   538→   538→  writeFileSync(ERRORS, JSON.stringify({ generatedAt: new Date().toISOString(), errors: errs }, null, 2), "utf8")
   539→   539→  await browser.close()
   540→   540→  return { snapshot, errors: errs }
   541→   541→}

   544→   544→function normRowValue(raw, flip) {
   545→   545→  if (raw === null || raw === undefined || !Number.isFinite(Number(raw))) return null
   546→   546→  let v = Number(raw)
   547→   547→  if (!Number.isFinite(v)) return null
   548→   548→  if (flip) v = -v
   549→   549→  if (v === 0) return null // zero placeholders are noise — skip (see policy header)
   550→   550→  if (Math.abs(v) > 1e15) return null // scale-insane
   551→   551→  return v
   552→   552→}

   554→   554→// Build a flat {metricCode: value} matrix per (statement, view, columnIndex)
   555→   555→function matrixFromView(view) {
   556→   556→  if (!view || view.error || !view.headers) return null
   557→   557→  const { periods, restated } = classifyView(view.headers, view.subheaders ?? [])
   558→   558→  const cells = new Map() // colIdx → Map(code → value)
   559→   559→  let dropped = 0
   560→   560→  for (const row of view.rows ?? []) {
   561→   561→    const rowMap = INCOME_ROWS[row.id] ?? BALANCE_ROWS[row.id] ?? CASH_ROWS[row.id]
   562→   562→    if (!rowMap) continue
   563→   563→    if (!Array.isArray(row.values) || row.values.length !== view.headers.length) { dropped++; continue }
   564→   564→    for (let i = 0; i < row.values.length; i++) {
   565→   565→      if (!periods[i]) continue
   566→   566→      const v = normRowValue(row.values[i], rowMap.flip)
   567→   567→      if (v === null) continue
   568→   568→      if (!cells.has(i)) cells.set(i, new Map())
   569→   569→      const m = cells.get(i)
   570→   570→      if (m.has(rowMap.code)) {
   571→   571→        // same metric twice in one view (unexpected) — keep the larger-magnitude row
   572→   572→        const prev = m.get(rowMap.code)
   573→   573→        m.set(rowMap.code, Math.abs(v) > Math.abs(prev) ? v : prev)
   574→   574→      } else m.set(rowMap.code, v)
   575→   575→    }
   576→   576→  }
   577→   577→  return { periods, cells, restated, dropped }
   578→   578→}

   597→   597→async function planFromSnapshot(snapshot, companies) {
   598→   598→  const companyIds = companies.map((c) => c.id)
   599→   599→  const existingReports = await loadExistingReports(companyIds)
   600→   600→  const existingValues = new Map() // reportId → Set(metricCode)
   601→   601→  const valRows = await db.financialValue.findMany({
   602→   602→    where: { companyId: { in: companyIds }, isDemoData: false },
   603→   603→    select: { reportId: true, metricCode: true, value: true, normalizedValue: true, subPeriod: true, id: true },
   604→   604→  })
   605→   605→  for (const v of valRows) {
   606→   606→    if (v.subPeriod && v.subPeriod !== "CURRENT") continue
   607→   607→    if (!existingValues.has(v.reportId)) existingValues.set(v.reportId, new Map())
   608→   608→    existingValues.get(v.reportId).set(v.metricCode, v.normalizedValue ?? v.value)
   609→   609→  }
   610→   610→
   611→   611→  const planEntries = []
   612→   612→  const conflicts = []
   613→   613→  const skipped = []
   614→   614→  const stats = { newReports: 0, fills: 0, conflicts: 0, skippedSlots: 0, companies: 0 }
   615→   615→  const sumChecks = [] // accumulated across all companies for the summary
   616→   616→
   617→   617→  for (const comp of companies) {
   618→   618→    const rec = snapshot.companies[comp.ticker]
   619→   619→    if (!rec) { skipped.push({ ticker: comp.ticker, reason: "NO_SNAPSHOT" }); continue }
   620→   620→    if (rec.notFound) { skipped.push({ ticker: comp.ticker, reason: "NO_SYMBOL_ON_TV" }); continue }
   621→   621→    const usable = Object.keys(rec.statements ?? {}).length
   622→   622→    if (usable === 0) { skipped.push({ ticker: comp.ticker, reason: "HARVEST_FAILED", errors: rec.errors }); continue }
   623→   623→    if (rec.currency && rec.currency.toUpperCase() !== "EGP") { skipped.push({ ticker: comp.ticker, reason: `CURRENCY_${rec.currency.toUpperCase()}` }); continue }
   624→   624→
   625→   625→    // company-level cross-check data: FY values per metric (from annual view) for sum-checks
   626→   626→    const perMetric = new Map() // metricCode → { annual: Map(fiscalYear→{value,end}), quarters: Map(label→{value,end,idx}) }
   627→   627→    const periodOwner = new Map() // "periodType|label" → { idx, end, statement, view, cells }
   628→   628→
   629→   629→    for (const [st, view] of Object.entries(rec.statements)) {
   630→   630→      for (const viewName of ["annual", "quarterly"]) {
   631→   631→        const m = matrixFromView(view[viewName])
   632→   632→        if (!m) continue
   633→   633→        m.periods.forEach((p, i) => {
   634→   634→          if (!p) return
   635→   635→          const key = `${p.periodType}|${p.periodLabel}`
   636→   636→          const cells = m.cells.get(i)
   637→   637→          if (!cells || cells.size === 0) return
   638→   638→          const prev = periodOwner.get(key)
   639→   639→          if (prev && daysBetween(prev.end, p.periodEnd) <= PERIOD_END_TOL_DAYS) {
   640→   640→            // same period already captured from another statement/view — merge metrics
   641→   641→            const target = prev.cells
   642→   642→            for (const [code, v] of cells) if (!target.has(code)) target.set(code, v)
   643→   643→            return
   644→   644→          }
   645→   645→          if (prev) {
   646→   646→            // same label but different periodEnd (fiscal-year change) — the later
   647→   647→            // statement in iteration order wins is ambiguous; prefer the one with
   648→   648→            // more metrics, log the ambiguity
   649→   649→            if (cells.size > prev.cells.size) periodOwner.set(key, { end: p.periodEnd, statement: st, view: viewName, cells, period: p, ambiguity: true })
   650→   650→            else prev.ambiguity = true
   651→   651→            return
   652→   652→          }
   653→   653→          periodOwner.set(key, { end: p.periodEnd, statement: st, view: viewName, cells, period: p })
   654→   654→        })
   655→   655→      }
   656→   656→    }
   657→   657→
   658→   658→    // fill perMetric for validation
   659→   659→    for (const [key, slot] of periodOwner.entries()) {
   660→   660→      for (const [code, v] of slot.cells) {
   661→   661→        if (!perMetric.has(code)) perMetric.set(code, { annual: new Map(), quarters: new Map() })
   662→   662→        const pm = perMetric.get(code)
   663→   663→        if (slot.period.periodType === "ANNUAL") pm.annual.set(slot.period.fiscalYear, { value: v, end: slot.period.periodEnd, label: slot.period.periodLabel, tvFiscalYear: slot.period.tvFiscalYear })
   664→   664→        else if (slot.period.periodType === "QUARTERLY") pm.quarters.set(key, { value: v, end: slot.period.periodEnd, label: slot.period.periodLabel, tvFiscalYear: slot.period.tvFiscalYear })
   665→   665→      }
   666→   666→    }
   667→   667→
   668→   668→    // sum-check: per metric, per TV fiscal year, Σ fiscal quarters vs FY.
   669→   669→    // TradingView quarterly labels carry the fiscal year (Q4 '18 = FY2018 for
   670→   670→    // a Jun-ending filer), so grouping by that year is exact — no date
   671→   671→    // windows, which would pull a 5th quarter for mid-year filers.
   672→   672→    const suspectSlots = new Set() // "periodType|label|code"
   673→   673→    const sumDetail = new Map() // suspect slot → {sum, fyValue, quarters}
   674→   674→    for (const [code, pm] of perMetric) {
   675→   675→      if (NO_SUM_CHECK.has(code)) continue // averages/ratios — summing is meaningless
   676→   676→      if (!FLOW_METRICS.has(code)) continue // stock (balance-sheet) metrics are point-in-time — never summed
   677→   677→      for (const [fy, fySlot] of pm.annual) {
   678→   678→        const qs = [...pm.quarters.entries()].filter(([, q]) => q.tvFiscalYear === fySlot.tvFiscalYear)
   679→   679→        if (qs.length < 3) continue // need most of the year for a meaningful check
   680→   680→        const sum = qs.reduce((a, [, q]) => a + q.value, 0)
   681→   681→        const tol = Math.max(Math.abs(fySlot.value) * SUM_TOL_PCT / 100, SUM_TOL_ABS)
   682→   682→        const ok = Math.abs(sum - fySlot.value) <= tol
   683→   683→        sumChecks.push({ code, fy, ok, sum, fyValue: fySlot.value, quarters: qs.length })
   684→   684→        if (!ok) for (const [k] of qs) { suspectSlots.add(`${k}|${code}`); sumDetail.set(`${k}|${code}`, { sum, fyValue: fySlot.value, quarters: qs.length }) }
   685→   685→      }
   686→   686→    }
   687→   687→
   688→   688→    // build plan entries per period
   689→   689→    for (const [key, slot] of periodOwner.entries()) {
   690→   690→      const { period } = slot
   691→   691→      const repKey = `${comp.id}|${period.periodType}|${period.periodLabel}|CONSOLIDATED`
   692→   692→      const existing = existingReports.get(repKey)
   693→   693→      const writes = []
   694→   694→      for (const [code, v] of slot.cells) {
   695→   695→        const suspect = suspectSlots.has(`${key}|${code}`)
   696→   696→        const kind = statementKind(code)
   697→   697→        if (kind === "OTHER" && !AR_LABEL[code]) { stats.skippedSlots++; continue }
   698→   698→        if (existing) {
   699→   699→          const slotVals = existingValues.get(existing.id)
   700→   700→          const prev = slotVals?.get(code)
   701→   701→          if (prev !== undefined) {
   702→   702→            const pct = Math.abs(v - prev) / Math.max(Math.abs(prev), 1) * 100
   703→   703→            if (pct > CONFLICT_PCT) { conflicts.push({ ticker: comp.ticker, period: period.periodLabel, metric: code, existing: prev, tv: v, diffPct: +pct.toFixed(1) }); stats.conflicts++ }
   704→   704→            continue // only-missing: never overwrite
   705→   705→          }
   706→   706→        }
   707→   707→        writes.push({ code, value: v, suspect, sumCheck: suspect ? sumDetail.get(`${key}|${code}`) : null })
   708→   708→      }
   709→   709→      if (writes.length === 0) continue
   710→   710→      // periodEnd guard on existing reports
   711→   711→      if (existing && existing.periodEnd && daysBetween(new Date(existing.periodEnd), period.periodEnd) > PERIOD_END_TOL_DAYS) {
   712→   712→        conflicts.push({
   713→   713→          ticker: comp.ticker, period: period.periodLabel, metric: "PERIOD",
   714→   714→          existing: new Date(existing.periodEnd).toISOString().slice(0, 10), tv: period.periodEnd.toISOString().slice(0, 10),
   715→   715→          diffPct: null, note: "PERIOD_END_MISMATCH (fiscal-year change) — skipped",
   716→   716→        })
   717→   717→        stats.conflicts++
   718→   718→        continue
   719→   719→      }
   720→   720→      planEntries.push({
   721→   721→        ticker: comp.ticker,
   722→   722→        companyId: comp.id,
   723→   723→        reportId: existing?.id ?? null,
   724→   724→        periodType: period.periodType,
   725→   725→        periodLabel: period.periodLabel,
   726→   726→        fiscalYear: period.fiscalYear,
   727→   727→        periodStart: new Date(period.periodEnd.getTime() - period.months * 30.44 * 86_400_000).toISOString().slice(0, 10),
   728→   728→        periodEnd: period.periodEnd.toISOString().slice(0, 10),
   729→   729→        statement: slot.statement,
   730→   730→        sourceUrl: rec.statements[slot.statement]?.url ?? null,
   731→   731→        tvSymbol: rec.tvSymbol,
   732→   732→        ambiguity: slot.ambiguity ?? false,
   733→   733→        values: writes.map((w) => ({ metric: w.code, value: w.value, kind: statementKind(w.code), label: AR_LABEL[w.code] ?? prettyCode(w.code), suspect: w.suspect, sumCheck: w.sumCheck })),
   734→   734→      })
   735→   735→      if (!existing) stats.newReports++
   736→   736→      stats.fills += writes.length
   737→   737→    }
   738→   738→    stats.companies++
   739→   739→  }
   740→   740→  return { entries: planEntries, conflicts, skipped, stats, sumChecks: sumChecksSample(sumChecks) }
   741→   741→}
   742→   742→function sumChecksSample(checks) {
   743→   743→  const bad = checks.filter((c) => !c.ok)
   744→   744→  return { total: checks.length, failed: bad.length, samples: bad.slice(0, 40) }
   745→   745→}

   748→   748→const STATUS_RANK = { APPROVED: 0, VALIDATED: 1, EXTRACTED: 2, PROCESSING: 3, NEW_DOWNLOADED: 4, NEEDS_REVIEW: 5, FAILED: 8, REJECTED: 9, DELETED: 10 }

   750→   750→async function applyPlan(entries) {
   751→   751→  mkdirSync("reports", { recursive: true })
   752→   752→  const rollback = { generatedAt: new Date().toISOString(), source: SOURCE, parser: PARSER, reports: [], values: [] }
   753→   753→  const blocked = []
   754→   754→  let reportCount = 0
   755→   755→  let valueCount = 0
   756→   756→
   757→   757→  // group by report slot: entries may target the same existing report from
   758→   758→  // different statements (income/balance/cash) — merge their values
   759→   759→  const bySlot = new Map()
   760→   760→  for (const e of entries) {
   761→   761→    const key = `${e.companyId}|${e.periodType}|${e.periodLabel}|CONSOLIDATED`
   762→   762→    if (!bySlot.has(key)) bySlot.set(key, { ...e, values: [...e.values] })
   763→   763→    else {
   764→   764→      const t = bySlot.get(key)
   765→   765→      const have = new Set(t.values.map((v) => v.metric))
   766→   766→      for (const v of e.values) if (!have.has(v.metric)) { t.values.push(v); have.add(v.metric) }
   767→   767→    }
   768→   768→  }
   769→   769→
   770→   770→  for (const entry of bySlot.values()) {
   771→   771→    let reportId = entry.reportId
   772→   772→    if (!reportId) {
   773→   773→      // soft-deleted / superseded / FAILED rows still occupy the key — never
   774→   774→      // resurrect a deliberately removed report (platform rule)
   775→   775→      const dup = await db.financialReport.findFirst({
   776→   776→        where: { companyId: entry.companyId, periodType: entry.periodType, periodLabel: entry.periodLabel, statementType: "CONSOLIDATED" },
   777→   777→        select: { id: true, deletedAt: true, processingStatus: true },
   778→   778→      })
   779→   779→      if (dup && (dup.deletedAt || (STATUS_RANK[dup.processingStatus] ?? 9) >= 8)) {
   780→   780→        blocked.push({ ticker: entry.ticker, periodLabel: entry.periodLabel, reason: dup.deletedAt ? "SOFT_DELETED" : dup.processingStatus })
   781→   781→        continue
   782→   782→      }
   783→   783→      if (dup) {
   784→   784→        reportId = dup.id
   785→   785→      } else {
   786→   786→        const created = await db.financialReport.create({
   787→   787→          data: {
   788→   788→            companyId: entry.companyId,
   789→   789→            reportType: entry.periodType === "ANNUAL" ? "ANNUAL" : "INTERIM",
   790→   790→            periodType: entry.periodType,
   791→   791→            statementType: "CONSOLIDATED",
   792→   792→            fiscalYear: entry.fiscalYear,
   793→   793→            periodLabel: entry.periodLabel,
   794→   794→            periodStart: new Date(entry.periodStart),
   795→   795→            periodEnd: new Date(entry.periodEnd),
   796→   796→            language: "EN",
   797→   797→            sourceName: SOURCE,
   798→   798→            sourceUrl: entry.sourceUrl,
   799→   799→            processingStatus: "EXTRACTED",
   800→   800→            extractionMethod: METHOD,
   801→   801→            parserVersion: PARSER,
   802→   802→            extractionConfidence: 0.85,
   803→   803→            isDemoData: false,
   804→   804→            version: 1,
   805→   805→            notes: `TradingView financial statements import (${entry.tvSymbol}); exact printed values via public symbol page.`,
   806→   806→          },
   807→   807→          select: { id: true },
   808→   808→        })
   809→   809→        reportId = created.id
   810→   810→        rollback.reports.push({ id: reportId, ticker: entry.ticker, periodLabel: entry.periodLabel })
   811→   811→        reportCount++
   812→   812→      }
   813→   813→    }
   814→   814→    // values: only-missing-slot with an in-memory pre-check + P2002 backstop
   815→   815→    const existing = await db.financialValue.findMany({
   816→   816→      where: { reportId },
   817→   817→      select: { metricCode: true, subPeriod: true },
   818→   818→    })
   819→   819→    const have = new Set(existing.filter((v) => !v.subPeriod || v.subPeriod === "CURRENT").map((v) => v.metricCode))
   820→   820→    for (const v of entry.values) {
   821→   821→      if (have.has(v.metric)) continue
   822→   822→      const validationStatus = v.suspect ? "NEEDS_REVIEW" : "VALID"
   823→   823→      const confidence = v.suspect ? 0.7 : 0.85
   824→   824→      const notes = [
   825→   825→        `${AUDIT_MARK} TradingView ${entry.tvSymbol} (${entry.statement}); source ${entry.sourceUrl}`,
   826→   826→        ...(v.suspect && v.sumCheck ? [`sum-check FAILED vs FY: Σ ${v.sumCheck.quarters} quarters = ${v.sumCheck.sum} vs FY ${v.sumCheck.fyValue} — flagged for review`] : []),
   827→   827→      ].join(" | ")
   828→   828→      try {
   829→   829→        const row = await db.financialValue.create({
   830→   830→          data: {
   831→   831→            companyId: entry.companyId,
   832→   832→            reportId,
   833→   833→            metricCode: v.metric,
   834→   834→            // originalLabel is user-visible (Arabic hint on the statements
   835→   835→            // table) — provenance lives in validationNotes (product rule).
   836→   836→            originalLabel: AR_LABEL[v.metric] ?? v.label ?? prettyCode(v.metric),
   837→   837→            value: v.value,
   838→   838→            currency: "EGP",
   839→   839→            unit: "UNIT",
   840→   840→            normalizedValue: v.value,
   841→   841→            statementType: v.kind || statementKind(v.metric),
   842→   842→            subPeriod: "CURRENT",
   843→   843→            extractionMethod: METHOD,
   844→   844→            confidence,
   845→   845→            validationStatus,
   846→   846→            validationNotes: notes,
   847→   847→            sourceText: entry.sourceUrl,
   848→   848→            isDemoData: false,
   849→   849→          },
   850→   850→          select: { id: true },
   851→   851→        })
   852→   852→        rollback.values.push({ id: row.id, ticker: entry.ticker, metric: v.metric, value: v.value })
   853→   853→        valueCount++
   854→   854→      } catch (e) {
   855→   855→        if (String(e?.code) === "P2002") { blocked.push({ ticker: entry.ticker, periodLabel: entry.periodLabel, metric: v.metric, reason: "VALUE_UNIQUE" }); continue }
   856→   856→        throw e
   857→   857→      }
   858→   858→    }
   859→   859→  }
   860→   860→  writeFileSync(ROLLBACK, JSON.stringify(rollback), "utf8")
   861→   861→  if (blocked.length) {
   862→   862→    writeFileSync("reports/tv-financials-blocked.json", JSON.stringify({ generatedAt: new Date().toISOString(), blocked }, null, 2), "utf8")
   863→   863→    log(`blocked (soft-deleted/failed/unique): ${blocked.length} → reports/tv-financials-blocked.json`)
   864→   864→  }
   865→   865→  await db.auditLog.create({
   866→   866→    data: {
   867→   867→      actor: "system",
   868→   868→      action: "UPLOAD",
   869→   869→      entityType: "FinancialReport",
   870→   870→      details: `${AUDIT_MARK} ${reportCount} reports created, ${valueCount} values added (parser ${PARSER}); rollback ${ROLLBACK}`,
   871→   871→    },
   872→   872→  })
   873→   873→  return { reportCount, valueCount }
   874→   874→}

   876→   876→async function runRollback() {
   877→   877→  if (!existsSync(ROLLBACK)) { console.error(`No rollback file at ${ROLLBACK}`); process.exit(1) }
   878→   878→  const data = JSON.parse(readFileSync(ROLLBACK, "utf8"))
   879→   879→  let values = 0
   880→   880→  let reports = 0
   881→   881→  let skipped = 0
   882→   882→  for (const v of data.values ?? []) {
   883→   883→    const row = await db.financialValue.findUnique({ where: { id: v.id }, select: { validationNotes: true } })
   884→   884→    if (!row || !String(row.validationNotes ?? "").includes(AUDIT_MARK)) { skipped++; continue }
   885→   885→    await db.financialValue.delete({ where: { id: v.id } })
   886→   886→    values++
   887→   887→  }
   888→   888→  for (const r of data.reports ?? []) {
   889→   889→    const rep = await db.financialReport.findUnique({ where: { id: r.id }, select: { notes: true, _count: { select: { values: true } } } })
   890→   890→    if (!rep || !String(rep.notes ?? "").includes("TradingView financial statements import")) { skipped++; continue }
   891→   891→    if (rep._count.values > 0) { skipped++; continue } // only empty reports; values were removed above
   892→   892→    await db.financialReport.delete({ where: { id: r.id } })
   893→   893→    reports++
   894→   894→  }
   895→   895→  await db.auditLog.create({
   896→   896→    data: {
   897→   897→      actor: "system",
   898→   898→      action: "DELETE",
   899→   899→      entityType: "FinancialReport",
   900→   900→      details: `${AUDIT_MARK} rollback: ${values} values removed, ${reports} reports deleted, ${skipped} skipped`,
   901→   901→    },
   902→   902→  })
   903→   903→  log(`rollback done: ${values} values, ${reports} reports, ${skipped} skipped`)
   904→   904→}

   941→   941→async function main() {
   942→   942→  if (ROLLBACK_MODE) { await runRollback(); await db.$disconnect(); return }
   943→   943→
   944→   944→  const companies = await db.company.findMany({
   945→   945→    where: { isActive: true, isDemoData: false },
   946→   946→    select: { id: true, ticker: true, nameEn: true, originalTicker: true },
   947→   947→    orderBy: { ticker: "asc" },
   948→   948→  })
   949→   949→  const target = ONLY_TICKERS ? companies.filter((c) => ONLY_TICKERS.includes(c.ticker.toUpperCase())) : companies.slice(0, LIMIT ?? companies.length)
   950→   950→  log(`target companies: ${target.length} (apply=${APPLY}, fromSnapshot=${FROM_SNAPSHOT}, workers=${WORKERS})`)
   951→   951→
   952→   952→  let snapshot
   953→   953→  if (FROM_SNAPSHOT) {
   954→   954→    snapshot = JSON.parse(readFileSync(SNAPSHOT, "utf8"))
   955→   955→    log(`using saved snapshot (${Object.keys(snapshot.companies).length} companies, generated ${snapshot.generatedAt})`)
   956→   956→  } else {
   957→   957→    // resume: a company is done when it has usable statements, or is known
   958→   958→    // absent from TradingView, or failed only with terminal errors. Anything
   959→   959→    // else (page errors, toggle failures) is retried on the next run.
   960→   960→    const prev = existsSync(SNAPSHOT) ? JSON.parse(readFileSync(SNAPSHOT, "utf8")) : { companies: {} }
   961→   961→    const isUsable = (rec) =>
   962→   962→      rec &&
   963→   963→      (rec.notFound ||
   964→   964→        (Object.keys(rec.statements ?? {}).length > 0 &&
   965→   965→          (rec.errors ?? []).every((e) => e.includes("NO_SYMBOL_ON_TV"))))
   966→   966→    const todo = target.filter((c) => !isUsable(prev.companies?.[c.ticker]))
   967→   967→    if (todo.length < target.length) log(`resume: ${target.length - todo.length} already harvested, ${todo.length} to (re)harvest`)
   968→   968→    const res = await runHarvest(todo)
   969→   969→    snapshot = res.snapshot
   970→   970→    const failed = res.errors.length
   971→   971→    const usable = Object.values(snapshot.companies).filter(isUsable).length
   972→   972→    log(`harvest finished: ${usable}/${target.length} usable, ${failed} with errors → ${ERRORS}`)
   973→   973→  }
   974→   974→
   975→   975→  if (HARVEST_ONLY) {
   976→   976→    log("harvest-only mode — no plan, no writes.")
   977→   977→    await db.$disconnect()
   978→   978→    return
   979→   979→  }
   980→   980→
   981→   981→  const plan = await planFromSnapshot(snapshot, target)
   982→   982→  writeFileSync(PLAN, JSON.stringify({ generatedAt: new Date().toISOString(), apply: APPLY, ...plan }, null, 2), "utf8")
   983→   983→  if (plan.conflicts.length) writeFileSync(CONFLICTS, JSON.stringify({ generatedAt: new Date().toISOString(), conflicts: plan.conflicts }, null, 2), "utf8")
   984→   984→  printPlanSummary(plan)
   985→   985→
   986→   986→  if (!APPLY) {
   987→   987→    log("dry run only — nothing written. Re-run with --apply to write.")
   988→   988→  } else {
   989→   989→    const t0 = Date.now()
   990→   990→    const res = await applyPlan(plan.entries)
   991→   991→    log(`apply done in ${((Date.now() - t0) / 1000).toFixed(0)}s: ${res.reportCount} reports created, ${res.valueCount} values written`)
   992→   992→    if (RECOMPUTE) await recomputeAll()
   993→   993→  }
   994→   994→  await db.$disconnect()
   995→   995→}
