    47→function HeaderSearch() {
    48→  const { lang } = useI18n()
    49→  const router = useRouter()
    50→  const [q, setQ] = useState("")
    51→  const [debounced, setDebounced] = useState("")
    52→  const [open, setOpen] = useState(false)
    53→  const [cursor, setCursor] = useState(0)
    54→  const boxRef = useRef<HTMLDivElement>(null)
    55→  const inputRef = useRef<HTMLInputElement>(null)
    56→  const ar = lang === "ar"
    57→
    58→  const popularTickers = [
    59→    { ticker: "COMI", nameEn: "Commercial International Bank", nameAr: "البنك التجاري الدولي", sector: "Banks" },
    60→    { ticker: "ETEL", nameEn: "Telecom Egypt", nameAr: "المصرية للاتصالات", sector: "Telecom" },
    61→    { ticker: "ABUK", nameEn: "Abu Qir Fertilizers", nameAr: "أبو قير للأسمدة", sector: "Basic Materials" },
    62→    { ticker: "TMGH", nameEn: "Talaat Moustafa Group", nameAr: "مجموعة طلعت مصطفى", sector: "Real Estate" },
    63→    { ticker: "EAST", nameEn: "Eastern Company", nameAr: "الشرقية للدخان", sector: "Consumer" },
    64→  ]
    65→
    66→  useEffect(() => {
    67→    const id = setTimeout(() => setDebounced(q.trim()), 200)
    68→    return () => clearTimeout(id)
    69→  }, [q])
    70→
    71→  useEffect(() => {
    72→    function onKeyDown(e: KeyboardEvent) {
    73→      if ((e.metaKey || e.ctrlKey) && e.key.toLowerCase() === "k") {
    74→        e.preventDefault()
    75→        inputRef.current?.focus()
    76→        setOpen(true)
    77→      }
    78→    }
    79→    window.addEventListener("keydown", onKeyDown)
    80→    return () => window.removeEventListener("keydown", onKeyDown)
    81→  }, [])
    82→
    83→  useEffect(() => {
    84→    function onDoc(e: MouseEvent) {
    85→      if (boxRef.current && !boxRef.current.contains(e.target as Node)) setOpen(false)
    86→    }
    87→    document.addEventListener("mousedown", onDoc)
    88→    return () => document.removeEventListener("mousedown", onDoc)
    89→  }, [])
    90→
    91→  const { data } = useQuery({
    92→    queryKey: ["header-search", debounced],
    93→    queryFn: () => api.companies({ search: debounced, pageSize: 6 }),
    94→    enabled: debounced.length >= 2,
    95→    staleTime: 60_000,
    96→  })
    97→  const results = debounced.length >= 2 ? data?.companies ?? [] : []
    98→
    99→  function go(id: string) {
   100→    setOpen(false)
   101→    setQ("")
   102→    router.push(`/company/${id}`)
   103→  }
   104→
   105→  const itemsToShow = debounced.length >= 2 ? results : open ? popularTickers : []
   106→
   107→  return (
   108→    <div ref={boxRef} className="relative">
   109→      <div className="flex items-center gap-1.5 rounded-lg border border-border/70 bg-muted/40 px-2.5 py-1.5 transition-all duration-150 focus-within:border-primary/60 focus-within:bg-background focus-within:ring-2 focus-within:ring-primary/20">
   110→        <Search className="h-3.5 w-3.5 shrink-0 text-muted-foreground" />
   111→        <input
   112→          ref={inputRef}
   113→          value={q}
   114→          onChange={(e) => {
   115→            setQ(e.target.value)
   116→            setOpen(true)
   117→            setCursor(0)
   118→          }}
   119→          onFocus={() => setOpen(true)}
   120→          onKeyDown={(e) => {
   121→            if (e.key === "ArrowDown") {
   122→              e.preventDefault()
   123→              setCursor((c) => Math.min(c + 1, itemsToShow.length - 1))
   124→            } else if (e.key === "ArrowUp") {
   125→              e.preventDefault()
   126→              setCursor((c) => Math.max(c - 1, 0))
   127→            } else if (e.key === "Enter" && itemsToShow[cursor]) {
   128→              e.preventDefault()
   129→              go((itemsToShow[cursor] as { id?: string; ticker: string }).id ?? itemsToShow[cursor].ticker)
   130→            } else if (e.key === "Escape") {
   131→              setOpen(false)
   132→              inputRef.current?.blur()
   133→            }
   134→          }}
   135→          placeholder={ar ? "ابحث عن شركة… (⌘K)" : "Search companies… (⌘K)"}
   136→          className="w-32 bg-transparent text-xs outline-none placeholder:text-muted-foreground/70 lg:w-48"
   137→          aria-label={ar ? "بحث الشركات" : "Company search"}
   138→        />
   139→        <kbd className="hidden lg:inline-flex items-center rounded border border-border/60 bg-muted/80 px-1 py-0.5 text-[9px] font-mono text-muted-foreground select-none">
   140→          ⌘K
   141→        </kbd>
   142→      </div>
   143→
   144→      {open && itemsToShow.length > 0 ? (
   145→        <div className="absolute -right-1 top-full z-50 mt-2 w-72 overflow-hidden rounded-xl border border-border/70 bg-card shadow-xl shadow-black/10 backdrop-blur-md animate-in fade-in-50 zoom-in-95 duration-100">
   146→          <div className="px-3 py-1.5 border-b border-border/40 text-[10px] font-medium text-muted-foreground uppercase tracking-wider flex items-center justify-between">
   147→            <span>{debounced.length >= 2 ? (ar ? "الشركات المطابقة" : "Matching companies") : (ar ? "شركات شائعة" : "Popular companies")}</span>
   148→            <span className="text-[9px]">{ar ? "اختر للفتح" : "Select to open"}</span>
   149→          </div>
   150→          <div className="max-h-72 overflow-y-auto scrollbar-thin py-1">
   151→            {itemsToShow.map((c, i) => (
   152→              <button
   153→                key={(c as { id?: string; ticker: string }).id ?? c.ticker}
   154→                onClick={() => go((c as { id?: string; ticker: string }).id ?? c.ticker)}
   155→                onMouseEnter={() => setCursor(i)}
   156→                className={`flex w-full items-center gap-2.5 px-3 py-2 text-start transition-colors ${
   157→                  i === cursor ? "bg-primary/10 text-primary" : "hover:bg-muted/50"
   158→                }`}
   159→              >
   160→                <span className="flex h-7 w-10 shrink-0 items-center justify-center rounded bg-primary/15 text-[10px] font-bold text-primary tabular">
   161→                  {c.ticker}
   162→                </span>
   163→                <span className="min-w-0 flex-1">
   164→                  <span className="block truncate text-xs font-semibold text-foreground">
   165→                    {lang === "ar" ? c.nameAr || c.nameEn : c.nameEn}
   166→                  </span>
   167→                  <span className="block text-[10px] text-muted-foreground">{c.sector}</span>
   168→                </span>
   169→              </button>
   170→            ))}
   171→          </div>
   172→        </div>
   173→      ) : null}
   174→    </div>
   175→  )
   176→}
