   153→   153→  const controller = new AbortController()
   154→   154→  const timer = setTimeout(() => controller.abort(), TIMEOUT_MS)
   155→   155→  try {
   156→   156→    const res = await fetch(url, {

   160→   160→    if (res.status === 404) return { status: 404, html: null }
   161→   161→    if (!res.ok) throw new Error(`HTTP ${res.status}`)
   162→   162→    return { status: 200, html: await res.text() }
   163→   163→  } finally {
   164→   164→    clearTimeout(timer)

   168→   168→async function fetchWithRetry(url, attempts = 3) {

   179→   179→async function mapLimit(items, limit, fn) {
   180→   180→  const out = new Array(items.length)
   181→   181→  let next = 0
   182→   182→  let done = 0
   183→   183→  async function worker() {
   184→   184→    while (true) {
   185→   185→      const i = next++
   186→   186→      if (i >= items.length) return
   187→   187→      out[i] = await fn(items[i])
   188→   188→      done++
   189→   189→      if (done % 50 === 0) log(`  progress ${done}/${items.length}`)
   190→   190→      await new Promise((r) => setTimeout(r, STAGGER_MS))
   191→   191→    }
   192→   192→  }
   193→   193→  await Promise.all(Array.from({ length: Math.min(limit, items.length) }, worker))
   194→   194→  return out
   195→   195→}

   202→   202→function extractPivots(html) {
   203→   203→  const text = unescapeNextFlight(html)
   204→   204→  const start = text.indexOf('[{"period":"')
   205→   205→  if (start < 0) return null
   206→   206→  let depth = 0
   207→   207→  let inString = false
   208→   208→  let escaped = false
   209→   209→  let end = -1
   210→   210→  for (let i = start; i < text.length; i++) {
   211→   211→    const ch = text[i]
   212→   212→    if (inString) {
   213→   213→      if (escaped) escaped = false
   214→   214→      else if (ch === "\\") escaped = true
   215→   215→      else if (ch === '"') inString = false
   216→   216→    } else {
   217→   217→      if (ch === '"') inString = true
   218→   218→      else if (ch === "[") depth++
   219→   219→      else if (ch === "]") {
   220→   220→        depth--
   221→   221→        if (depth === 0) { end = i + 1; break }
   222→   222→      }
   223→   223→    }
   224→   224→  }
   225→   225→  if (end < 0) return null
   226→   226→  try {
   227→   227→    const pivots = JSON.parse(text.slice(start, end))
   228→   228→    if (!Array.isArray(pivots)) return null
   229→   229→    return pivots.filter((p) => p && typeof p.period === "string" && Array.isArray(p.rows))
   230→   230→  } catch {
   231→   231→    return null
   232→   232→  }
   233→   233→}

   248→   248→function periodHrefPeriods(html, ticker) {
   249→   249→  const escapedTicker = ticker.replace(/[.*+?^${}()|[\]\\]/g, "\\$")
   250→   250→  const re = new RegExp(`href="/company/${escapedTicker}/(fy|q[1-4])-(\\d{4})"`, "g")
   251→   251→  const out = new Set()
   252→   252→  let m

   254→   254→  return [...out]
   255→   255→}

   257→   257→// Statement page: metric blocks in the flight payload.
   258→   258→//   hero card : ["$","div","{id}",{"className":"card ...
   259→   259→//   table row : ["$","div","{id}",{"className":"grid grid-cols... (current, prior, delta)
   260→   260→function parseStatementPage(html, ticker) {
   261→   261→  // each push wraps one JS string literal; match it atomically so an escaped
   262→   262→  // quote or an inner "]" never truncates the chunk
   263→   263→  const chunks = [...html.matchAll(/self\.__next_f\.push\(\[1,"((?:[^"\\]|\\.)*)"\]\)/g)].map((m) => m[1])
   264→   264→  if (!chunks.length) return { metrics: {}, otherPeriods: [], parsed: false }
   265→   265→  const flight = unescapeNextFlight(chunks.join(""))
   266→   266→
   267→   267→  // block boundaries: every ["$","div","<id>",{"className":"card|grid...
   268→   268→  const marks = []
   269→   269→  const markRe = /\["\$","div","([a-z_0-9]+)",\{"className":"(?:card|grid)/g
   270→   270→  let mm
   271→   271→  while ((mm = markRe.exec(flight))) marks.push({ id: mm[1], start: mm.index })
   272→   272→  const metrics = {}
   273→   273→  for (let i = 0; i < marks.length; i++) {
   274→   274→    const segEnd = i + 1 < marks.length ? marks[i + 1].start : Math.min(marks[i].start + 4000, flight.length)
   275→   275→    const seg = flight.slice(marks[i].start, segEnd)
   276→   276→    const metric = ITEM_TO_METRIC[marks[i].id]
   277→   277→    if (!metric) continue
   278→   278→    const titles = [...seg.matchAll(/"title":"([0-9][0-9,.-]*)"/g)].map((m) => Number(m[1].replace(/,/g, "")))
   279→   279→    const isHero = /className":"card/.test(seg.slice(0, 80))
   280→   280→    const entry = metrics[metric] ?? {}
   281→   281→    if (titles[0] !== undefined && Number.isFinite(titles[0])) entry.current = titles[0]
   282→   282→    if (!isHero && titles[1] !== undefined && Number.isFinite(titles[1])) entry.prior = titles[1]
   283→   283→    metrics[metric] = entry
   284→   284→  }
   285→   285→  return { metrics, otherPeriods: periodHrefPeriods(html, ticker), parsed: true }
   286→   286→}

   311→   311→async function harvest(tickers) {
   312→   312→  log(`harvest: fetching ${tickers.length} overview pages`)
   313→   313→  const overviews = await mapLimit(tickers, CONCURRENCY, async (ticker) => {
   314→   314→    const url = `https://askborsa.com/company/${encodeURIComponent(ticker)}`
   315→   315→    const res = await fetchWithRetry(url)
   316→   316→    if (res.status === 404) return { ticker, status: "not_found" }
   317→   317→    if (res.status !== 200) return { ticker, status: "fetch_failed", error: res.error ?? `HTTP ${res.status}` }
   318→   318→    const meta = extractOverviewMeta(res.html)
   319→   319→    const pivots = extractPivots(res.html)
   320→   320→    const periods = periodHrefPeriods(res.html, ticker)
   321→   321→    return { ticker, status: "ok", url, ...meta, pivots: pivots ?? [], overviewPeriods: periods }
   322→   322→  })
   323→   323→
   324→   324→  // build statement-page fetch plan
   325→   325→  const fetchPlan = new Map() // ticker -> Set(periodKey "q3-2026")
   326→   326→  for (const o of overviews) {
   327→   327→    if (o.status !== "ok") continue
   328→   328→    const set = new Set()
   329→   329→    const covered = new Set()
   330→   330→    for (const p of o.pivots ?? []) {
   331→   331→      for (const y of p.years ?? []) {
   332→   332→        const key = `${p.period.toLowerCase()}-${y}`
   333→   333→        covered.add(key)
   334→   334→        // pivot-covered periods still get a page fetch when the pivot row set
   335→   335→        // is thin (pivot lacks stock/cf rows for that period) — skip for now;
   336→   336→        // comparatives from newer pages backfill older years anyway.
   337→   337→      }
   338→   338→    }
   339→   339→    for (const key of o.overviewPeriods ?? []) if (!covered.has(key)) set.add(key)
   340→   340→    // generated candidates: years beyond each pivot window up to CURRENT_YEAR
   341→   341→    const maxByType = {}
   342→   342→    for (const p of o.pivots ?? []) {
   343→   343→      const t = p.period.toLowerCase()
   344→   344→      for (const y of p.years ?? []) maxByType[t] = Math.max(maxByType[t] ?? 0, y)
   345→   345→    }
   346→   346→    for (const t of PERIOD_TYPES) {
   347→   347→      // no pivot at all → probe a bounded recent window (2023+) and let BFS find older
   348→   348→      const from = maxByType[t] ? maxByType[t] + 1 : 2023
   349→   349→      for (let y = Math.max(from, 2019); y <= CURRENT_YEAR; y++) {
   350→   350→        const key = `${t}-${y}`
   351→   351→        if (!covered.has(key)) set.add(key)
   352→   352→      }
   353→   353→    }
   354→   354→    fetchPlan.set(o.ticker, set)
   355→   355→  }
   356→   356→
   357→   357→  let pages = [...fetchPlan.entries()].flatMap(([t, keys]) => [...keys].map((k) => ({ ticker: t, key: k })))

   454→   454→// ---------------------------------------------------------------- pattern votes
   455→   455→// Per (ticker, metric): vote "discrete" | "cumulative" | null from years where
   456→   456→// the full flow series + FY value exist on AskBorsa.
   457→   457→function votePatterns(snapshot) {
   458→   458→  const votes = new Map() // ticker -> metric -> { discrete, cumulative, years: [] }
   459→   459→  for (const r of snapshot.results) {
   460→   460→    if (r.status !== "ok") continue
   461→   461→    // collect per fiscal year flow series: values for q1..q4 + fy
   462→   462→    const byYear = new Map()
   463→   463→    for (const [key, bucket] of Object.entries(r.periods)) {
   464→   464→      const [t, yStr] = key.split("-")
   465→   465→      const y = Number(yStr)
   466→   466→      for (const metric of FLOW) {
   467→   467→        const v = bucket[metric]?.value
   468→   468→        if (v === undefined || !Number.isFinite(v) || v === 0) continue
   469→   469→        if (!byYear.has(y)) byYear.set(y, {})
   470→   470→        const rec = byYear.get(y)
   471→   471→        ;(rec[t] ??= {})[metric] = v
   472→   472→      }
   473→   473→    }
   474→   474→    for (const [y, rec] of byYear) {
   475→   475→      const fyVals = rec.fy ?? {}
   476→   476→      for (const metric of FLOW) {
   477→   477→        const fy = fyVals[metric]
   478→   478→        if (fy === undefined) continue
   479→   479→        const qs = ["q1", "q2", "q3", "q4"].map((t) => rec[t]?.[metric])
   480→   480→        if (qs.some((v) => v === undefined)) continue // need full series
   481→   481→        const [q1, q2, q3, q4] = qs
   482→   482→        let vote = null
   483→   483→        if (closeSum(q1 + q2 + q3 + q4, fy) && ![q1, q2, q3].some((v) => approx(v, fy)) && !([q1, q2, q3].every((v) => v === 0))) {
   484→   484→          vote = "discrete"
   485→   485→        } else {
   486→   486→          // cumulative: same sign, |magnitudes| non-decreasing, terminal ≈ FY
   487→   487→          const series = [q1, q2, q3, q4]
   488→   488→          const signs = series.map(sameSign)
   489→   489→          const sign = signs.find((s) => s !== 0)
   490→   490→          const monotone = sign !== undefined && signs.every((s) => s === sign || s === 0)
   491→   491→            && series.every((v, i) => i === 0 || Math.abs(v) >= Math.abs(series[i - 1]) * 0.995)
   492→   492→          const terminalOk = approx(q4, fy) && Math.abs(q4 / fy - 1) <= 0.01
   493→   493→          const accumulation = Math.abs(q3) >= Math.abs(q1) * 1.2
   494→   494→          if (monotone && terminalOk && accumulation) vote = "cumulative"
   495→   495→        }
   496→   496→        if (vote) {
   497→   497→          if (!votes.has(r.ticker)) votes.set(r.ticker, {})
   498→   498→          const v = votes.get(r.ticker)
   499→   499→          v[metric] ??= { discrete: 0, cumulative: 0, years: [] }
   500→   500→          v[metric][vote]++
   501→   501→          v[metric].years.push(`${y}:${vote}`)
   502→   502→        }
   503→   503→      }
   504→   504→    }
   505→   505→  }
   506→   506→  const patterns = new Map()
   507→   507→  for (const [ticker, byMetric] of votes) {
   508→   508→    for (const [metric, v] of Object.entries(byMetric)) {
   509→   509→      const winner = v.discrete === v.cumulative ? null : (v.discrete > v.cumulative ? "discrete" : "cumulative")
   510→   510→      patterns.set(`${ticker}|${metric}`, { pattern: winner, discrete: v.discrete, cumulative: v.cumulative })
   511→   511→    }
   512→   512→  }
   513→   513→  return patterns
   514→   514→}

   713→   713→              const cum = cumulativeCheck(rec, metric, fy)
   714→   714→              if (cum) {
   715→   715→                const tgt = { q1: ["QUARTERLY", `Q1 ${y}`], q2: ["SEMIANNUAL", `H1 ${y}`], q3: ["NINE_MONTH", `9M ${y}`], q4: null }[t]
   716→   716→                if (tgt) { target = { pt: tgt[0], label: tgt[1] }; verifiedBy = "CUMULATIVE_SERIES" }
   717→   717→              }
   718→   718→            }
   719→   719→            if (!target && !fy) skip(r.ticker, qRec.key, metric, "FLOW_GATE_FAILED_HISTORICAL")
   720→   720→            if (!target && fy) { /* discrete failed & cumulative failed for a metric with FY: only flag when full series existed */
   721→   721→              const full = ["q1", "q2", "q3", "q4"].every((tt) => rec[tt]?.bucket[metric]?.value !== undefined)
   722→   722→              if (full) skip(r.ticker, qRec.key, metric, "FLOW_GATE_FAILED_FULL_SERIES")
   723→   723→            }
   724→   724→          } else {
   725→   725→            // current year: historical pattern + monotonicity + prior-year ratio
   726→   726→            const pat = patterns.get(`${r.ticker}|${metric}`)
   727→   727→            if (!pat?.pattern) { skip(r.ticker, qRec.key, metric, "FLOW_NO_PATTERN"); continue }
   728→   728→            const priorVal = r.periods[`${t}-${y - 1}`]?.[metric]?.value
   729→   729→            if (priorVal === undefined) { skip(r.ticker, qRec.key, metric, "FLOW_NO_PRIOR_YEAR"); continue }
   730→   730→            const ratio = Math.abs(v) / Math.abs(priorVal)
   731→   731→            if (!(ratio >= 0.05 && ratio <= 20)) { skip(r.ticker, qRec.key, metric, `FLOW_PRIOR_RATIO_SUSPECT (${ratio.toFixed(2)}x)`); continue }
   732→   732→            const mono = cumulativeMonotoneUpTo(rec, metric, t)
   733→   733→            if (mono === false) { skip(r.ticker, qRec.key, metric, "FLOW_MONOTONICITY_BROKEN"); continue }
   734→   734→            if (pat.pattern === "discrete") {
   735→   735→              target = { pt: "QUARTERLY", label: `${t.toUpperCase()} ${y}` }
   736→   736→              verifiedBy = "PATTERN_DISCRETE"
   737→   737→            } else if (pat.pattern === "cumulative") {
   738→   738→              if (t === "q4") { skip(r.ticker, qRec.key, metric, "CUMULATIVE_Q4_COVERED_BY_FY"); continue }
   739→   739→              const tgt = { q1: ["QUARTERLY", `Q1 ${y}`], q2: ["SEMIANNUAL", `H1 ${y}`], q3: ["NINE_MONTH", `9M ${y}`] }[t]
   740→   740→              target = { pt: tgt[0], label: tgt[1] }
   741→   741→              verifiedBy = "PATTERN_CUMULATIVE"
   742→   742→            }
   743→   743→          }
   744→   744→          if (!target) continue
   745→   745→          const tk = `${target.pt}|${target.label}`
   746→   746→          const arr = flowByTarget.get(tk) ?? []
   747→   747→          arr.push({ metric, value: v, verifiedBy })
   748→   748→          flowByTarget.set(tk, arr)
   749→   749→        }
   750→   750→
   751→   751→        // emit plans for this quarter
   752→   752→        const targets = new Map()
   753→   753→        if (stockValues.length) targets.set(`QUARTERLY|${t.toUpperCase()} ${y}`, { pt: "QUARTERLY", label: `${t.toUpperCase()} ${y}`, values: stockValues })
   754→   754→        for (const [tk, vals] of flowByTarget) {
   755→   755→          const [pt, label] = tk.split("|")
   756→   756→          const ex = targets.get(tk)
   757→   757→          if (ex) ex.values.push(...vals)
   758→   758→          else targets.set(tk, { pt, label, values: vals })
   759→   759→        }
   760→   760→        for (const [tk, spec] of targets) {
   761→   761→          const existingReport = per.get(reportKey(spec.pt, spec.label))
   762→   762→          // per-value sanity vs the app's own annuals of the same metric.
   763→   763→          // flows get a wider band than stocks: seasonal quarters can be a
   764→   764→          // tiny fraction of the year (GPIM Q1 ≈ 4% of FY), while a balance
   765→   765→          // sheet cannot plausibly collapse to 5% year-over-year. Real unit
   766→   766→          // errors are ~1000x and are blocked by both bands.
   767→   767→          const saneValues = []
   768→   768→          for (const v of spec.values) {
   769→   769→            const ref = nearestAnnual(annualRef.get(v.metric), y)
   770→   770→            if (ref === undefined) { saneValues.push(v); continue }
   771→   771→            const ratio = Math.abs(v.value) / Math.abs(ref)
   772→   772→            const [lo, hi] = FLOW.has(v.metric) ? [0.01, 50] : [0.05, 20]
   773→   773→            if (ratio >= lo && ratio <= hi) saneValues.push(v)
   774→   774→            else skip(r.ticker, qRec.key, v.metric, `VALUE_SCALE_VS_APP (${ratio.toExponential(1)}x vs ${v.metric} annual)`)
   775→   775→          }
   776→   776→          if (!saneValues.length) continue
   777→   777→          spec.values = saneValues
   778→   778→          // a Q4 report carrying ONLY point-in-time stock values would just
   779→   779→          // duplicate the annual balance sheet — never create it
   780→   780→          if (!existingReport && t === "q4" && !spec.values.some((v) => FLOW.has(v.metric))) {
   781→   781→            skip(r.ticker, qRec.key, null, "Q4_STOCK_ONLY_NO_REPORT")
   782→   782→            continue
   783→   783→          }
   784→   784→          if (!existingReport) {
   785→   785→            plan.push({
   786→   786→              action: "CREATE_REPORT", kind: "QUARTER", ticker: r.ticker, companyId: company.id,
   787→   787→              periodType: spec.pt, periodLabel: spec.label, fiscalYear: y,
   788→   788→              periodStart: quarterStart(end).toISOString(), periodEnd: end.toISOString(),
   789→   789→              sourceUrl: r.sourceUrl, confidence: spec.values.some((v) => String(v.verifiedBy).startsWith("PATTERN")) ? 0.7 : 0.8,
   790→   790→              verifiedBy: spec.values.map((v) => v.verifiedBy).join("+"),
   791→   791→              values: spec.values,
   792→   792→            })
   793→   793→          } else {
   794→   794→            const have = new Set(existingReport.values
   795→   795→              .filter((v) => v.subPeriod === "CURRENT" || v.subPeriod === null)
   796→   796→              .map((v) => v.metricCode))
   797→   797→            const fills = spec.values.filter((v) => !have.has(v.metric))
   798→   798→            if (fills.length) {
   799→   799→              plan.push({
   800→   800→                action: "ADD_VALUE", kind: "QUARTER", ticker: r.ticker, companyId: company.id,
   801→   801→                reportId: existingReport.id, reportKey: spec.label, fiscalYear: y,
   802→   802→                sourceUrl: r.sourceUrl, confidence: fills.some((v) => String(v.verifiedBy).startsWith("PATTERN")) ? 0.7 : 0.8,
   803→   803→                verifiedBy: fills.map((v) => v.verifiedBy).join("+"),
   804→   804→                values: fills,
   805→   805→              })
   806→   806→            }
   807→   807→            for (const v of spec.values) {
   808→   808→              const exV = existingReport.values.find((x) => x.metricCode === v.metric && (x.subPeriod === "CURRENT" || x.subPeriod === null) && Number.isFinite(x.normalizedValue))
   809→   809→              if (exV) {
   810→   810→                const d = relDiff(v.value, exV.normalizedValue)
   811→   811→                if (d !== null && d > 0.05) conflicts.push({ ticker: r.ticker, period: spec.label, metric: v.metric, askValue: v.value, appValue: exV.normalizedValue, relDiff: Number(d.toFixed(4)), askSource: "askborsa" })
   812→   812→              }
   813→   813→            }
   814→   814→          }
   815→   815→        }
   816→   816→      }
   817→   817→    }
   818→   818→  }
   819→   819→
   820→   820→  return { plan, skipped, conflicts }
   821→   821→}

   823→   823→function nearestAnnual(arr, year) {
   824→   824→  if (!arr?.length) return undefined
   825→   825→  let best
   826→   826→  for (const a of arr) {
   827→   827→    const dist = Math.abs(a.year - year)
   828→   828→    if (!best || dist < best.dist) best = { dist, value: a.value }
   829→   829→  }
   830→   830→  return best.value
   831→   831→}

   833→   833→// cumulative check for a year with ask FY: series Q1..Q3 (+Q4 or FY as terminal)
   834→   834→function cumulativeCheck(rec, metric, fy) {
   835→   835→  const vals = ["q1", "q2", "q3"].map((t) => rec[t]?.bucket[metric]?.value)
   836→   836→  if (vals.some((v) => v === undefined || v === 0)) return false
   837→   837→  const q4 = rec.q4?.bucket[metric]?.value
   838→   838→  const series = q4 !== undefined && q4 !== 0 ? [...vals, q4] : [...vals, fy]
   839→   839→  if (series.some((v) => v === undefined || v === 0)) return false
   840→   840→  const sign = sameSign(series[0])
   841→   841→  if (sign === 0) return false
   842→   842→  const monotone = series.every((v) => sameSign(v) === sign) && series.every((v, i) => i === 0 || Math.abs(v) >= Math.abs(series[i - 1]) * 0.995)
   843→   843→  const terminalOk = q4 !== undefined && q4 !== 0 ? approx(q4, fy) && Math.abs(q4 / fy - 1) <= 0.01 : true
   844→   844→  const accumulation = Math.abs(vals[2]) >= Math.abs(vals[0]) * 1.2
   845→   845→  // must NOT pass the discrete sum gate (that case is handled first)
   846→   846→  const allFour = ["q1", "q2", "q3", "q4"].map((t) => rec[t]?.bucket[metric]?.value)
   847→   847→  const discretePass = allFour.every((v) => v !== undefined) && closeSum(allFour.reduce((a, b) => a + b, 0), fy)
   848→   848→  return monotone && terminalOk && accumulation && !discretePass
   849→   849→}
   850→   850→
   851→   851→// monotonicity of cumulative pattern up to quarter t (for current-year import)
   852→   852→function cumulativeMonotoneUpTo(rec, metric, upTo) {
   853→   853→  const idx = { q1: 1, q2: 2, q3: 3, q4: 4 }[upTo]
   854→   854→  const series = []
   855→   855→  for (const t of ["q1", "q2", "q3", "q4"].slice(0, idx)) {
   856→   856→    const v = rec[t]?.bucket[metric]?.value
   857→   857→    if (v !== undefined && v !== 0) series.push(v)
   858→   858→  }
   859→   859→  if (series.length < 2) return true // nothing to contradict
   860→   860→  const sign = sameSign(series[0])
   861→   861→  return sign !== 0 && series.every((v) => sameSign(v) === sign) && series.every((v, i) => i === 0 || Math.abs(v) >= Math.abs(series[i - 1]) * 0.995)
   862→   862→}

   864→   864→// ---------------------------------------------------------------- apply
   865→   865→async function applyPlan(plan) {
   866→   866→  mkdirSync("reports", { recursive: true })
   867→   867→  const rollback = { generatedAt: new Date().toISOString(), reports: [], values: [] }
   868→   868→  let reportCount = 0
   869→   869→  let valueCount = 0
   870→   870→  for (const p of plan) {

   873→   873→      const dup = await db.financialReport.findFirst({
   874→   874→        where: { companyId: p.companyId, periodType: p.periodType, periodLabel: p.periodLabel, statementType: "CONSOLIDATED", deletedAt: null, isDemoData: false },
   875→   875→        select: { id: true },
   876→   876→      })
   877→   877→      if (dup) {
   878→   878→        reportId = dup.id
   879→   879→      } else {
   880→   880→        const created = await db.financialReport.create({
   881→   881→          data: {
   882→   882→            companyId: p.companyId,
   883→   883→            reportType: p.periodType === "ANNUAL" ? "ANNUAL" : "INTERIM",
   884→   884→            periodType: p.periodType,
   885→   885→            statementType: "CONSOLIDATED",
   886→   886→            fiscalYear: p.fiscalYear,
   887→   887→            periodLabel: p.periodLabel,
   888→   888→            periodStart: new Date(p.periodStart),
   889→   889→            periodEnd: new Date(p.periodEnd),
   890→   890→            language: "AR",
   891→   891→            sourceName: "ASKBORA_PUBLIC",
   892→   892→            sourceUrl: p.sourceUrl,
   893→   893→            processingStatus: "EXTRACTED",
   894→   894→            extractionMethod: "ASKBORA_HARVEST",
   895→   895→            parserVersion: PARSER,
   896→   896→            extractionConfidence: p.confidence,
   897→   897→            isDemoData: false,
   898→   898→            version: 1,
   899→   899→            notes: `AskBorsa public data sync; verified: ${p.verifiedBy}.`,
   900→   900→          },
   901→   901→          select: { id: true },
   902→   902→        })
   903→   903→        reportId = created.id
   904→   904→        rollback.reports.push({ id: reportId, ticker: p.ticker, periodLabel: p.periodLabel })
   905→   905→        reportCount++
   906→   906→      }
   907→   907→    } else {
   908→   908→      reportId = p.reportId
   909→   909→    }
   910→   910→    for (const v of p.values) {
   911→   911→      // final only-missing-slot re-check under concurrency-safe conditions
   912→   912→      const slot = await db.financialValue.findFirst({
   913→   913→        where: { reportId, metricCode: v.metric, OR: [{ subPeriod: "CURRENT" }, { subPeriod: null }] },
   914→   914→        select: { id: true },
   915→   915→      })
   916→   916→      if (slot) continue
   917→   917→      const row = await db.financialValue.create({
   918→   918→        data: {
   919→   919→          companyId: p.companyId,
   920→   920→          reportId,
   921→   921→          metricCode: v.metric,
   922→   922→          // Product rule: originalLabel is user-visible (bilingual hint on the
   923→   923→          // statements table) — never embed tooling/source refs like
   924→   924→          // "(AskBorsa Qx)" here. Provenance lives in validationNotes instead.
   925→   925→          originalLabel: AR_LABEL[v.metric] ?? EN_FALLBACK[v.metric] ?? prettyCode(v.metric),
   926→   926→          value: v.value,
   927→   927→          currency: "EGP",
   928→   928→          unit: "UNIT",
   929→   929→          normalizedValue: v.value,
   930→   930→          statementType: statementKind(v.metric),
   931→   931→          subPeriod: "CURRENT",
   932→   932→          extractionMethod: "ASKBORA_HARVEST",
   933→   933→          confidence: p.confidence ?? 0.8,
   934→   934→          validationStatus: "VALID",
   935→   935→          validationNotes: `${MARK} ${v.verifiedBy}; source ${p.sourceUrl}`,
   936→   936→          sourceText: p.sourceUrl,
   937→   937→          isDemoData: false,
   938→   938→        },
   939→   939→        select: { id: true },
   940→   940→      })
   941→   941→      rollback.values.push({ id: row.id, ticker: p.ticker, metric: v.metric, value: v.value })
   942→   942→      valueCount++
   943→   943→    }
   944→   944→  }
   945→   945→  writeFileSync(ROLLBACK, JSON.stringify(rollback, null, 2), "utf8")
   946→   946→  await db.auditLog.create({
   947→   947→    data: {
   948→   948→      actor: "system",
   949→   949→      action: "UPLOAD",
   950→   950→      entityType: "FinancialReport",
   951→   951→      details: `ASKBORA_SYNC: ${reportCount} reports created, ${valueCount} values added (parser ${PARSER}); rollback ${ROLLBACK}`,
   952→   952→    },
   953→   953→  })
   954→   954→  return { reportCount, valueCount }
   955→   955→}

   957→   957→async function runRollback() {
   958→   958→  if (!existsSync(ROLLBACK)) { console.error(`No rollback file at ${ROLLBACK}`); process.exit(1) }
   959→   959→  const data = JSON.parse(readFileSync(ROLLBACK, "utf8"))
   960→   960→  let values = 0
   961→   961→  let reports = 0
   962→   962→  let skipped = 0
   963→   963→  for (const v of data.values ?? []) {
   964→   964→    const row = await db.financialValue.findUnique({ where: { id: v.id }, select: { validationNotes: true } })
   965→   965→    if (!row || !String(row.validationNotes ?? "").startsWith(MARK)) { skipped++; continue }
   966→   966→    await db.financialValue.delete({ where: { id: v.id } })
   967→   967→    values++
   968→   968→  }
   969→   969→  for (const r of data.reports ?? []) {
   970→   970→    const row = await db.financialReport.findUnique({ where: { id: r.id }, select: { sourceName: true, _count: { select: { values: true } } } })
   971→   971→    if (!row || row.sourceName !== "ASKBORA_PUBLIC" || row._count.values > 0) { skipped++; continue }
   972→   972→    await db.financialReport.delete({ where: { id: r.id } })
   973→   973→    reports++
   974→   974→  }
   975→   975→  console.log(`rollback complete: values=${values}, reports=${reports}, skipped=${skipped}`)
   976→   976→}

   999→   999→async function main() {
  1000→  1000→  if (ROLLBACK_MODE) return runRollback()
  1001→  1001→
  1002→  1002→  mkdirSync("reports", { recursive: true })
  1003→  1003→
  1004→  1004→  let snapshot
  1005→  1005→  if (FROM_SNAPSHOT && existsSync(SNAPSHOT)) {
  1006→  1006→    snapshot = JSON.parse(readFileSync(SNAPSHOT, "utf8"))
  1007→  1007→    log(`loaded snapshot ${SNAPSHOT} (${snapshot.generatedAt})`)
  1008→  1008→  } else {
  1009→  1009→    const dbCompanies = await db.company.findMany({
  1010→  1010→      where: { isActive: true, ...(ONLY_TICKERS ? { ticker: { in: ONLY_TICKERS } } : {}) },
  1011→  1011→      select: { ticker: true },
  1012→  1012→      orderBy: { ticker: "asc" },
  1013→  1013→    })
  1014→  1014→    let tickers = dbCompanies.map((c) => c.ticker)
  1015→  1015→    if (LIMIT) tickers = tickers.slice(0, LIMIT)
  1016→  1016→    snapshot = await harvest(tickers)
  1017→  1017→    writeFileSync(SNAPSHOT, JSON.stringify(snapshot, null, 2), "utf8")
  1018→  1018→    log(`wrote ${SNAPSHOT}`)
  1019→  1019→  }
  1020→  1020→
  1021→  1021→  const patterns = votePatterns(snapshot)
  1022→  1022→  const discreteCount = [...patterns.values()].filter((p) => p.pattern === "discrete").length
  1023→  1023→  const cumulativeCount = [...patterns.values()].filter((p) => p.pattern === "cumulative").length
  1024→  1024→  log(`flow patterns voted: ${patterns.size} (ticker|metric) pairs — discrete=${discreteCount}, cumulative=${cumulativeCount}`)
  1025→  1025→
  1026→  1026→  const { plan, skipped, conflicts } = await buildPlan(snapshot, patterns)
  1027→  1027→  const createReports = plan.filter((p) => p.action === "CREATE_REPORT")
  1028→  1028→  const addValues = plan.filter((p) => p.action === "ADD_VALUE")
  1029→  1029→  const totalValues = plan.reduce((n, p) => n + p.values.length, 0)
  1030→  1030→  const byTicker = {}
  1031→  1031→  for (const p of plan) byTicker[p.ticker] = (byTicker[p.ticker] ?? 0) + 1
  1032→  1032→  const skipReasons = {}
  1033→  1033→  for (const s of skipped) skipReasons[s.reason ?? "UNKNOWN"] = (skipReasons[s.reason ?? "UNKNOWN"] ?? 0) + 1
  1034→  1034→
  1035→  1035→  log(`PLAN: ${createReports.length} reports to create, ${addValues.length} slot-fills, ${totalValues} values total, ${skipped.length} skips, ${conflicts.length} conflicts`)
  1036→  1036→  log(`top companies: ${Object.entries(byTicker).sort((a, b) => b[1] - a[1]).slice(0, 10).map(([t, n]) => `${t}(${n})`).join(", ")}`)
  1037→  1037→  log(`skip reasons: ${JSON.stringify(skipReasons)}`)
  1038→  1038→
  1039→  1039→  writeFileSync(PLAN, JSON.stringify({ generatedAt: new Date().toISOString(), mode: APPLY ? "apply" : "dry-run", createReports: createReports.length, addValues: addValues.length, totalValues, plan, skipped }, null, 2), "utf8")
  1040→  1040→  log(`wrote ${PLAN}`)
  1041→  1041→  if (conflicts.length) {
  1042→  1042→    writeFileSync(CONFLICTS, JSON.stringify({ generatedAt: new Date().toISOString(), conflicts }, null, 2), "utf8")
  1043→  1043→    log(`wrote ${CONFLICTS} (${conflicts.length} conflicts for review — no data written for them)`)
  1044→  1044→  }
  1045→  1045→
  1046→  1046→  if (!APPLY) {
  1047→  1047→    log("DRY RUN — rerun with --apply to write. Add --recompute to rebuild derived metrics after import.")
  1048→  1048→    return
  1049→  1049→  }
  1050→  1050→
  1051→  1051→  const res = await applyPlan(plan)
  1052→  1052→  log(`APPLIED: reports=${res.reportCount}, values=${res.valueCount}; rollback saved to ${ROLLBACK}`)
  1053→  1053→
  1054→  1054→  if (RECOMPUTE) await runRecompute()
  1055→  1055→  else log("next step: node scripts/run-recompute.cjs  (or rerun with --recompute)")
  1056→  1056→}
