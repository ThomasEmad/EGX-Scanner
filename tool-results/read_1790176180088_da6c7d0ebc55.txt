   156→function OverviewTab({ companyId }: { companyId: string }) {
   157→  const { t, lang } = useI18n()
   158→  const { data: metrics, isLoading } = useQuery({ queryKey: ["metrics", companyId], queryFn: () => api.metrics(companyId) })
   159→
   160→  if (isLoading || !metrics) return <Skeleton className="h-64 rounded-xl" />
   161→
   162→  const annualPeriods = metrics.periods.filter((p) => p.periodType === "ANNUAL").sort((a, b) => a.fiscalYear - b.fiscalYear)
   163→  const latest = annualPeriods[annualPeriods.length - 1]
   164→
   165→  const getCell = (code: string) => (latest ? metrics.rows.find((r) => r.code === code)?.cells[latest.key] : undefined)
   166→
   167→  const cards: { code: string; label: string; fmt: (v: number) => string; invert?: boolean }[] = [
   168→    { code: "revenue", label: t("co.revenue"), fmt: (v) => formatEgp(v) },
   169→    { code: "net_profit", label: t("co.netProfit"), fmt: (v) => formatEgp(v), invert: true },
   170→    { code: "roe", label: t("co.roe"), fmt: (v) => formatPercent(v) },
   171→    { code: "total_assets", label: t("co.assets"), fmt: (v) => formatEgp(v) },
   172→    { code: "debt_to_equity", label: t("co.growth") + " D/E", fmt: (v) => formatRatio(v) },
   173→    { code: "eps", label: "EPS", fmt: (v) => `EGP ${v.toFixed(2)}` },
   174→  ]
   175→
   176→  // trend chart data (revenue & net profit over annual periods)
   177→  const revenueRow = metrics.rows.find((r) => r.code === "revenue")
   178→  const profitRow = metrics.rows.find((r) => r.code === "net_profit")
   179→  const trend = annualPeriods.map((p) => ({
   180→    label: p.label.replace("FY ", ""),
   181→    revenue: revenueRow?.cells[p.key]?.value ?? null,
   182→    profit: profitRow?.cells[p.key]?.value ?? null,
   183→  }))
   184→
   185→  return (
   186→    <div className="space-y-5">
   187→      <div className="grid grid-cols-2 gap-3 sm:grid-cols-3 lg:grid-cols-6">
   188→        {cards.map((card) => {
   189→          const cell = getCell(card.code)
   190→          const unavailable = !cell || cell.status !== "OK" || cell.value === null
   191→          return (
   192→            <Card key={card.code} className="p-3">
   193→              <p className="text-[11px] text-muted-foreground leading-tight">{card.label}</p>
   194→              {unavailable ? (
   195→                <TooltipProvider delayDuration={0}>
   196→                  <Tooltip>
   197→                    <TooltipTrigger asChild>
   198→                      <p className="mt-1.5 flex items-center gap-1 text-sm font-semibold text-slate-400">
   199→                        — <Info className="h-3 w-3" />
   200→                      </p>
   201→                    </TooltipTrigger>
   202→                    <TooltipContent side="top" className="max-w-56 text-[11px]">
   203→                      {cell?.detail || t("sc.dataUnavailable")}
   204→                    </TooltipContent>
   205→                  </Tooltip>
   206→                </TooltipProvider>
   207→              ) : (
   208→                <p className={`mt-1.5 text-sm font-bold tabular ${card.invert && (cell.value as number) < 0 ? "text-red-600 dark:text-red-400" : ""}`}>
   209→                  {card.fmt(cell.value as number)}
   210→                </p>
   211→              )}
   212→              {latest ? <p className="mt-0.5 text-[10px] text-muted-foreground">{latest.label}</p> : null}
   213→            </Card>
   214→          )
   215→        })}
   216→      </div>
   217→
   218→      {/* market snapshot (only when an observed price point exists) */}
   219→      <MarketSnapshot metrics={metrics} companyId={companyId} />
   220→
   221→      {/* trend chart */}
   222→      <Card className="p-0">
   223→        <CardHeader className="pb-2">
   224→          <CardTitle className="text-sm">{lang === "ar" ? "اتجاه الإيرادات وصافي الربح (سنوي)" : "Revenue & net profit trend (annual, EGP)"}</CardTitle>
   225→        </CardHeader>
   226→        <CardContent>
   227→          <TrendChart data={trend} />
   228→        </CardContent>
   229→      </Card>
   230→    </div>
   231→  )
   232→}

   234→/* ---------------- Market snapshot ---------------- */

   236→function MarketSnapshot({ metrics, companyId }: { metrics: MetricsData; companyId: string }) {
   237→  const { t } = useI18n()
   238→  const { data: detail } = useQuery({ queryKey: ["company", companyId], queryFn: () => api.company(companyId) })
   239→
   240→  // prefer the latest TTM window (most current trailing fundamentals), else latest annual
   241→  const ttmPeriods = metrics.periods.filter((p) => p.periodType === "TTM").sort((a, b) => a.key.localeCompare(b.key))
   242→  const annualPeriods = metrics.periods.filter((p) => p.periodType === "ANNUAL").sort((a, b) => a.key.localeCompare(b.key))
   243→  const basis = ttmPeriods[ttmPeriods.length - 1] ?? annualPeriods[annualPeriods.length - 1]
   244→  if (!basis) return null
   245→
   246→  const getCell = (code: string) => metrics.rows.find((r) => r.code === code)?.cells[basis.key]
   247→
   248→  const cards: { code: string; label: string; fmt: (v: number) => string }[] = [
   249→    { code: "market_cap", label: t("co.marketCap"), fmt: (v) => formatEgp(v) },
   250→    { code: "p_b", label: t("co.pb"), fmt: (v) => formatRatio(v) },
   251→    { code: "p_e", label: t("co.pe"), fmt: (v) => formatRatio(v) },
   252→    { code: "dividend_yield", label: t("co.yield"), fmt: (v) => formatPercent(v) },
   253→  ]
   254→
   255→  const mp = detail?.marketPrice ?? null
   256→
   257→  return (
   258→    <Card className="p-0">
   259→      <CardHeader className="pb-2 flex flex-row flex-wrap items-center justify-between gap-2">
   260→        <CardTitle className="flex items-center gap-1.5 text-sm">
   261→          <TrendingUp className="h-4 w-4 text-primary" />
   262→          {t("co.market")}
   263→        </CardTitle>
   264→        {mp ? (
   265→          <TooltipProvider delayDuration={0}>
   266→            <Tooltip>
   267→              <TooltipTrigger asChild>
   268→                <span className="flex items-baseline gap-1.5 rounded-lg border border-primary/30 bg-primary/5 px-2.5 py-1 cursor-help">
   269→                  <span className="text-[10px] uppercase tracking-wide text-muted-foreground">{t("co.price")}</span>
   270→                  <span className="text-sm font-bold tabular text-primary">
   271→                    {mp.price.toFixed(2)} <span className="text-[10px] font-normal">{mp.currency}</span>
   272→                  </span>
   273→                  <span className="text-[10px] text-muted-foreground">
   274→                    {t("co.asOf")} {new Date(mp.asOf).toLocaleDateString("en-GB", { day: "numeric", month: "short", year: "numeric" })}
   275→                  </span>
   276→                </span>
   277→              </TooltipTrigger>
   278→              <TooltipContent side="top" className="max-w-56 text-[11px]">
   279→                {mp.sourceName}
   280→                {mp.isDemoData ? ` — ${t("co.demoPrice")}` : ""}
   281→              </TooltipContent>
   282→            </Tooltip>
   283→          </TooltipProvider>
   284→        ) : null}
   285→      </CardHeader>
   286→      <CardContent className="pt-0">
   287→        <div className="grid grid-cols-2 gap-3 lg:grid-cols-4">
   288→          {cards.map((card) => {
   289→            const cell = getCell(card.code)
   290→            const unavailable = !cell || cell.status !== "OK" || cell.value === null
   291→            return (
   292→              <div key={card.code} className="rounded-lg border bg-muted/30 p-3">
   293→                <p className="text-[10px] uppercase tracking-wide text-muted-foreground">{card.label}</p>
   294→                {unavailable ? (
   295→                  <TooltipProvider delayDuration={0}>
   296→                    <Tooltip>
   297→                      <TooltipTrigger asChild>
   298→                        <p className="mt-1 flex items-center gap-1 text-sm font-semibold text-slate-400">
   299→                          — <Info className="h-3 w-3" />
   300→                        </p>
   301→                      </TooltipTrigger>
   302→                      <TooltipContent side="top" className="max-w-64 text-[11px]">
   303→                        {cell?.detail || t("sc.dataUnavailable")}
   304→                      </TooltipContent>
   305→                    </Tooltip>
   306→                  </TooltipProvider>
   307→                ) : (
   308→                  <p className="mt-1 text-sm font-bold tabular">{card.fmt(cell.value as number)}</p>
   309→                )}
   310→                <p className="mt-0.5 text-[9px] text-muted-foreground/70">{cell?.formulaVersion ?? ""}</p>
   311→              </div>
   312→            )
   313→          })}
   314→        </div>
   315→      </CardContent>
   316→    </Card>
   317→  )
   318→}

   320→function TrendChart({ data }: { data: { label: string; revenue: number | null; profit: number | null }[] }) {
   321→  const max = Math.max(...data.flatMap((d) => [d.revenue ?? 0, d.profit ?? 0]), 1)
   322→  return (
   323→    <div className="flex items-end gap-6 sm:gap-10 px-2 pt-4 pb-1" dir="ltr">
   324→      {data.map((d) => (
   325→        <div key={d.label} className="flex flex-1 flex-col items-center gap-1">
   326→          <div className="flex h-36 w-full max-w-16 items-end justify-center gap-1.5">
   327→            <div className="relative flex h-full w-6 items-end">
   328→              <div
   329→                className="w-full rounded-t bg-primary/70 transition-all"
   330→                style={{ height: `${Math.max(2, ((d.revenue ?? 0) / max) * 100)}%` }}
   331→                title={`Revenue: ${formatEgp(d.revenue)}`}
   332→              />
   333→            </div>
   334→            <div className="relative flex h-full w-6 items-end">
   335→              <div
   336→                className={`w-full rounded-t transition-all ${(d.profit ?? 0) < 0 ? "bg-red-400" : "bg-emerald-400"}`}
   337→                style={{ height: `${Math.max(2, (Math.abs(d.profit ?? 0) / max) * 100)}%` }}
   338→                title={`Net profit: ${formatEgp(d.profit)}`}
   339→              />
   340→            </div>
   341→          </div>
   342→          <span className="text-[11px] font-medium text-muted-foreground">{d.label}</span>
   343→        </div>
   344→      ))}
   345→    </div>
   346→  )
   347→}

   427→function GrowthTab({ companyId }: { companyId: string }) {
   428→  const { t, lang } = useI18n()
   429→  const { data, isLoading } = useQuery({ queryKey: ["metrics", companyId], queryFn: () => api.metrics(companyId) })
   430→
   431→  if (isLoading || !data) return <Skeleton className="h-72 rounded-xl" />
   432→
   433→  const periods = [...data.periods].sort((a, b) => a.key.localeCompare(b.key))
   434→  const interesting = data.rows.filter((r) => r.kind !== "passthrough")
   435→
   436→  const fmtCell = (code: string, value: number | null) => {
   437→    const row = data.rows.find((r) => r.code === code)
   438→    if (value === null) return null
   439→    if (row?.unit === "PERCENT") return formatPercent(value)
   440→    if (row?.unit === "RATIO") return formatRatio(value)
   441→    if (row?.unit === "EGP_PER_SHARE") return `EGP ${value.toFixed(2)}`
   442→    if (row?.unit === "EGP") return formatEgp(value, { withCurrency: false })
   443→    return value.toFixed(2)
   444→  }
   445→
   446→  return (
   447→    <Card className="p-0">
   448→      <CardHeader className="pb-2">
   449→        <CardTitle className="text-sm">{t("co.growth")}</CardTitle>
   450→      </CardHeader>
   451→      <CardContent className="p-0">
   452→        <div className="max-h-[520px] overflow-auto scrollbar-thin">
   453→          <table className="w-full text-xs">
   454→            <thead className="sticky top-0 bg-card">
   455→              <tr className="border-b">
   456→                <th className="p-2.5 text-start font-medium text-muted-foreground min-w-48">{lang === "ar" ? "المؤشر" : "Metric"}</th>
   457→                {periods.map((p) => (
   458→                  <th key={p.key} className="p-2.5 text-end font-medium text-muted-foreground whitespace-nowrap">{p.label}</th>
   459→                ))}
   460→              </tr>
   461→            </thead>
   462→            <tbody>
   463→              {interesting.map((row) => (
   464→                <tr key={row.code} className="border-b last:border-0 hover:bg-muted/40">
   465→                  <td className="p-2.5 font-medium">
   466→                    {lang === "ar" ? row.labelAr : row.labelEn}
   467→                    <span className="ms-1.5 text-[9px] text-muted-foreground/70">{row.cells[Object.keys(row.cells)[0]]?.formulaVersion ?? ""}</span>
   468→                  </td>
   469→                  {periods.map((p) => {
   470→                    const cell = row.cells[p.key]
   471→                    const text = cell ? fmtCell(row.code, cell.value) : null
   472→                    return (
   473→                      <td key={p.key} className="p-2.5 text-end tabular whitespace-nowrap">
   474→                        {cell && cell.status === "OK" && text ? (
   475→                          <span className={/^[-]/.test(text.trim()) ? "text-red-600 dark:text-red-400" : ""}>{text}</span>
   476→                        ) : (
   477→                          <TooltipProvider delayDuration={0}>
   478→                            <Tooltip>
   479→                              <TooltipTrigger asChild>
   480→                                <span className="cursor-help text-slate-400">—</span>
   481→                              </TooltipTrigger>
   482→                              <TooltipContent side="top" className="max-w-56 text-[11px]">
   483→                                {cell?.detail || t("sc.dataUnavailable")}
   484→                              </TooltipContent>
   485→                            </Tooltip>
   486→                          </TooltipProvider>
   487→                        )}
   488→                      </td>

   491→                </tr>

   493→            </tbody>
   494→          </table>
